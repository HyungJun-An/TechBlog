export function AutoConfigFlowViz() {
  const [step, setStep] = useState(-1);
  const [running, setRunning] = useState(false);
  const [showConditions, setShowConditions] = useState(false);
  const timerRef = useRef(null);

  const STEPS = [
    { icon: '🚀', label: 'SpringApplication.run() 호출', desc: '메인 클래스에서 애플리케이션이 시작됩니다.', color: '#6366f1' },
    { icon: '📄', label: 'AutoConfiguration.imports 로드', desc: 'META-INF/spring/ 경로의 자동 설정 목록 파일을 읽습니다. Spring Boot 2.7+ 기준 (~150개 설정 클래스 목록).', color: '#3b82f6' },
    { icon: '🔎', label: '@Conditional 조건 평가', desc: '각 자동 설정 클래스의 조건 어노테이션을 평가합니다. 클래스패스에 특정 라이브러리가 있는지, 빈이 이미 있는지 등을 확인합니다.', color: '#f59e0b' },
    { icon: '✅', label: '조건 충족 → 빈 등록', desc: '조건을 통과한 설정 클래스의 @Bean 메서드가 실행되어 빈이 컨테이너에 등록됩니다.', color: '#10b981' },
    { icon: '🫘', label: 'ApplicationContext 완성', desc: '자동 등록 빈 + 사용자 빈이 모두 합쳐진 컨테이너가 완성됩니다. 이후 요청 처리 준비 완료.', color: '#8b5cf6' },
  ];

  const CONDITIONS = [
    { ann: '@ConditionalOnClass', desc: '특정 클래스가 클래스패스에 존재할 때', example: 'DataSource.class가 있으면 DB 자동 설정' },
    { ann: '@ConditionalOnMissingBean', desc: '해당 타입 빈이 아직 등록되지 않았을 때', example: '사용자가 직접 DataSource를 빈으로 만들면 자동 설정 스킵' },
    { ann: '@ConditionalOnProperty', desc: 'application.properties 특정 키 값이 조건과 일치할 때', example: 'spring.security.enabled=true 이면 활성화' },
    { ann: '@ConditionalOnWebApplication', desc: '웹 애플리케이션 컨텍스트일 때', example: 'SERVLET / REACTIVE 타입 구분 가능' },
  ];

  const start = () => {
    clearInterval(timerRef.current);
    setStep(0);
    setRunning(true);
    let i = 0;
    timerRef.current = setInterval(() => {
      i++;
      if (i >= STEPS.length) { clearInterval(timerRef.current); setRunning(false); setStep(STEPS.length - 1); return; }
      setStep(i);
    }, 1000);
  };

  const reset = () => { clearInterval(timerRef.current); setStep(-1); setRunning(false); };
  useEffect(() => () => clearInterval(timerRef.current), []);

  const wrap = { margin: '32px 0', borderRadius: '18px', overflow: 'hidden', border: '1px solid var(--ifm-color-emphasis-300)', background: 'var(--ifm-background-surface-color)' };
  const btnBase = { padding: '7px 18px', border: 'none', borderRadius: '8px', fontWeight: '700', fontSize: '13px', cursor: 'pointer', transition: 'all 0.2s' };

  return (
    <div style={wrap}>
      <div style={{ padding: '16px 24px', borderBottom: '1px solid var(--ifm-color-emphasis-200)' }}>
        <div style={{ fontWeight: '700', fontSize: '15px' }}>🤖 @EnableAutoConfiguration 동작 흐름</div>
      </div>
      <div style={{ padding: '24px' }}>
        <div style={{ display: 'flex', flexDirection: 'column', gap: '10px', marginBottom: '20px' }}>
          {STEPS.map((s, i) => (
            <div key={i} style={{ display: 'flex', gap: '12px', alignItems: 'flex-start', padding: '14px 16px', borderRadius: '10px', border: `1px solid ${step >= i ? s.color + '44' : 'var(--ifm-color-emphasis-200)'}`, background: step >= i ? s.color + '0d' : 'transparent', transition: 'all 0.4s', opacity: step >= i ? 1 : 0.35 }}>
              <div style={{ fontSize: '20px', marginTop: '1px', flexShrink: 0 }}>{s.icon}</div>
              <div>
                <div style={{ fontWeight: '700', fontSize: '13px', color: step >= i ? s.color : 'var(--ifm-color-content-secondary)' }}>{i + 1}. {s.label}</div>
                {step >= i && <div style={{ fontSize: '12px', color: 'var(--ifm-color-content-secondary)', marginTop: '4px', lineHeight: '1.6' }}>{s.desc}</div>}
              </div>
            </div>
          ))}
        </div>

        <div style={{ display: 'flex', gap: '8px', marginBottom: '20px' }}>
          <button onClick={start} disabled={running} style={{ ...btnBase, background: running ? 'var(--ifm-color-emphasis-300)' : '#3b82f6', color: '#fff', opacity: running ? 0.6 : 1 }}>▶ 실행</button>
          <button onClick={reset} style={{ ...btnBase, background: 'var(--ifm-color-emphasis-200)', color: 'var(--ifm-color-content)' }}>↺ 초기화</button>
          <button onClick={() => setShowConditions(!showConditions)} style={{ ...btnBase, background: showConditions ? '#f59e0b' : 'var(--ifm-color-emphasis-200)', color: showConditions ? '#fff' : 'var(--ifm-color-content)' }}>@Conditional 목록</button>
        </div>

        {showConditions && (
          <div style={{ display: 'flex', flexDirection: 'column', gap: '8px' }}>
            <div style={{ fontWeight: '700', fontSize: '13px', marginBottom: '4px', color: '#f59e0b' }}>자주 쓰이는 @Conditional 어노테이션</div>
            {CONDITIONS.map((c, i) => (
              <div key={i} style={{ padding: '12px 14px', borderRadius: '8px', border: '1px solid rgba(245,158,11,0.25)', background: 'rgba(245,158,11,0.06)' }}>
                <div style={{ fontFamily: 'monospace', fontSize: '12px', fontWeight: '700', color: '#f59e0b', marginBottom: '4px' }}>{c.ann}</div>
                <div style={{ fontSize: '12px', color: 'var(--ifm-color-content-secondary)' }}>{c.desc}</div>
                <div style={{ fontSize: '11px', color: 'var(--ifm-color-content-secondary)', marginTop: '3px', fontStyle: 'italic' }}>예: {c.example}</div>
              </div>
            ))}
          </div>
        )}
      </div>
    </div>
  );
}

`@SpringBootApplication`은 스프링 부트 프로젝트 어디에나 있지만, 그 안에 세 개의 어노테이션이 합성되어 있다는 사실을 아는 개발자는 많지 않습니다. 자동 설정이 어떤 원리로 동작하는지, 스테레오타입 어노테이션들이 진짜로 다른 점이 무엇인지를 정리합니다.

## @SpringBootApplication의 정체

메인 클래스에 붙는 이 어노테이션은 실제로 세 개의 어노테이션이 합쳐진 합성 어노테이션입니다.

```java
// @SpringBootApplication 내부 정의 (요약)
@SpringBootConfiguration     // = @Configuration의 부트 특화 버전
@EnableAutoConfiguration      // 자동 설정 활성화
@ComponentScan                // 베이스 패키지 하위 컴포넌트 스캔
public @interface SpringBootApplication { ... }
```

어노테이션을 클릭해서 각각의 역할을 확인하세요.

<SpringBootAppViz />

### 베이스 패키지 규칙 — 가장 흔한 초기 오류

`@ComponentScan`은 **메인 클래스가 위치한 패키지와 그 하위**만 탐색합니다.

```
com.example.myapp
├── MyAppApplication.java   ← @SpringBootApplication (베이스 패키지)
├── user/
│   └── UserService.java    ✅ 스캔됨
└── order/
    └── OrderService.java   ✅ 스캔됨

com.other.utils
└── SharedHelper.java       ❌ 베이스 패키지 밖 → 스캔 안 됨 → 주입 시 오류
```

외부 패키지까지 포함하려면 `scanBasePackages`를 명시합니다.

```java
@SpringBootApplication(scanBasePackages = "com.example")
```

## @EnableAutoConfiguration — 자동 설정의 원리

"의존성 추가 한 줄로 DB, 보안, MVC가 자동으로 설정되는" 마법의 배후는 조건부 빈 등록입니다.

<AutoConfigFlowViz />

### 핵심 원리: "조건이 맞으면만 등록한다"

자동 설정은 무조건 모든 빈을 등록하지 않습니다. 각 설정 클래스에 `@Conditional` 조건이 붙어 있어, 클래스패스나 기존 빈 상태를 보고 등록 여부를 결정합니다.

```java
// spring-boot-autoconfigure 내부 코드 (WebMvcAutoConfiguration 일부)
@AutoConfiguration
@ConditionalOnWebApplication(type = SERVLET)         // 서블릿 웹 앱일 때만
@ConditionalOnClass({ Servlet.class, DispatcherServlet.class }) // 이 클래스가 있을 때만
@ConditionalOnMissingBean(WebMvcConfigurationSupport.class)     // 직접 설정 안 했을 때만
public class WebMvcAutoConfiguration { ... }
```

이 덕분에 **사용자가 직접 `DataSource` 빈을 등록하면 자동 설정이 스킵**됩니다. 자동 설정은 기본값일 뿐, 직접 정의한 빈이 항상 우선합니다.

> **디버깅 팁:** `--debug` 플래그 또는 `logging.level.org.springframework.boot.autoconfigure=DEBUG`로 실행하면 어떤 자동 설정이 적용/거부되었는지 콘솔에서 확인할 수 있습니다.

## 빈 등록 방식 — @Configuration + @Bean vs 스테레오타입

빈을 등록하는 방법은 크게 두 갈래입니다.

| 방식 | 어노테이션 | 등록 단위 | 적합한 상황 |
|---|---|---|---|
| **설정 클래스 기반** | `@Configuration` + `@Bean` | 메서드 반환 객체 | 외부 라이브러리, 같은 타입 여러 인스턴스 |
| **스테레오타입 기반** | `@Component` 계열 | 클래스 자체 | 직접 작성한 내부 컴포넌트 |

### 스테레오타입 계층 구조

```
@Component                    ← 최상위 (범용)
├── @Controller               ← 웹 요청 처리
│   └── @RestController       ← + @ResponseBody
├── @Service                  ← 비즈니스 로직
└── @Repository               ← 데이터 접근 + 예외 변환
```

모두 `@Component`의 확장이므로 컴포넌트 스캔 대상이 됩니다. 차이점을 탭으로 비교해보세요.

<StereotypeCompare />

### 실무 판단 기준

- **내가 직접 작성한 클래스** → `@Service`, `@Repository`, `@Controller` 등 역할에 맞는 스테레오타입
- **외부 라이브러리 클래스를 빈으로 만들 때** → `@Configuration + @Bean` (소스를 수정할 수 없으므로)
- **같은 타입을 여러 이름으로 구분할 때** → `@Configuration + @Bean(name = "...")`

## 다중 빈 주입 해결 — @Qualifier vs @Resource

같은 타입 빈이 2개 이상 있으면 `@Autowired`만으로는 어느 빈을 주입할지 결정할 수 없어 애플리케이션이 **시작 단계에서** 실패합니다.

```
NoUniqueBeanDefinitionException: expected single matching bean but found 2: acOn, acOff
```

<QualifierResourceCompare />

### 한 줄 요약

| 어노테이션 | 탐색 순서 | 출처 | 실무 빈도 |
|---|---|---|---|
| `@Autowired + @Qualifier` | 타입 → 이름 | Spring | ★★★★★ |
| `@Resource(name = "...")` | 이름 → 타입 | Jakarta EE | ★★☆☆☆ |

> **추가 옵션:** `@Autowired(required = false)` — 대상 빈이 없어도 `null`로 주입하고 애플리케이션 기동을 허용합니다. 빈이 조건부로 존재하는 레거시 환경에서 방어 코드로 사용합니다.

## 정리

```
@SpringBootApplication
├── @SpringBootConfiguration  → 메인 클래스가 설정 클래스
├── @EnableAutoConfiguration  → 클래스패스 분석 + @Conditional 조건 충족 빈 자동 등록
└── @ComponentScan            → 베이스 패키지 하위 @Component 계열 자동 탐지·등록

빈 등록 방식
├── @Configuration + @Bean    → 메서드 단위, 외부 라이브러리, 다중 인스턴스
└── 스테레오타입
    ├── @Component             → 범용
    ├── @Controller            → 웹 요청 처리 (HandlerMapping 탐지)
    ├── @Service               → 비즈니스 로직 (의미론적 구분)
    └── @Repository            → 데이터 접근 + DB 예외 → DataAccessException 자동 변환

다중 빈 주입
├── @Autowired + @Qualifier   → 타입 먼저, 이름으로 한정 (스프링 표준)
└── @Resource(name = "...")   → 이름 먼저, 타입 fallback (Jakarta EE 표준)
```

다음 글에서는 **DI 주입 방식 4가지** — 생성자 주입, 필드 주입, Setter 주입, 메서드 주입의 동작 방식과 실무 권장 순서를 다룹니다.
