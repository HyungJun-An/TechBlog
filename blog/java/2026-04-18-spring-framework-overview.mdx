import { useState, useEffect, useRef } from 'react';

{/* ────────────────────────────────────────────────────────────
    Component 1: 서블릿 vs 프론트컨트롤러 요청 흐름 시각화
──────────────────────────────────────────────────────────── */}
export function RequestFlowViz() {
  const [mode, setMode] = useState('servlet');
  const [step, setStep] = useState(-1);
  const [running, setRunning] = useState(false);
  const timerRef = useRef(null);

  const servletSteps = [
    { label: '요청 도착', desc: 'HTTP 요청이 웹 서버에 들어옵니다.', color: '#6366f1' },
    { label: '서블릿 매핑 탐색', desc: 'web.xml 또는 @WebServlet 으로 URL → 서블릿 클래스를 직접 찾습니다.', color: '#f59e0b' },
    { label: 'UserServlet 실행', desc: '요청별 서블릿이 request 파싱 + 형 변환 + 비즈니스 로직을 모두 처리합니다.', color: '#ef4444' },
    { label: '뷰 경로 직접 조합', desc: '"/WEB-INF/views/" + viewName + ".jsp" 를 수동으로 포워드합니다.', color: '#ec4899' },
    { label: '응답 반환', desc: '처리 완료. 그런데 기능이 100개면 서블릿 클래스도 100개...', color: '#94a3b8' },
  ];

  const springSteps = [
    { label: '요청 도착', desc: 'HTTP 요청이 웹 서버에 들어옵니다.', color: '#6366f1' },
    { label: 'DispatcherServlet (단일 진입점)', desc: '모든 요청이 단 하나의 프론트 컨트롤러에서 수신됩니다.', color: '#3b82f6' },
    { label: 'HandlerMapping → 핸들러 탐색', desc: '@RequestMapping 기반으로 어떤 컨트롤러 메서드를 실행할지 결정합니다.', color: '#10b981' },
    { label: 'HandlerAdapter → 컨트롤러 실행', desc: '파라미터 바인딩·형 변환을 자동 처리한 뒤 메서드를 호출합니다.', color: '#f59e0b' },
    { label: 'ViewResolver → 뷰 선택', desc: '뷰 이름("home")만 반환하면 prefix/suffix를 자동으로 조합합니다.', color: '#8b5cf6' },
    { label: '응답 반환', desc: '컨트롤러는 비즈니스 로직에만 집중. 공통 처리는 프레임워크가 담당합니다.', color: '#14b8a6' },
  ];

  const steps = mode === 'servlet' ? servletSteps : springSteps;

  const start = () => {
    clearInterval(timerRef.current);
    setStep(0);
    setRunning(true);
    let i = 0;
    timerRef.current = setInterval(() => {
      i++;
      if (i >= steps.length) { clearInterval(timerRef.current); setRunning(false); setStep(steps.length - 1); return; }
      setStep(i);
    }, 900);
  };

  const reset = () => { clearInterval(timerRef.current); setStep(-1); setRunning(false); };

  useEffect(() => () => clearInterval(timerRef.current), []);
  useEffect(() => { reset(); }, [mode]);

  const wrap = { margin: '32px 0', borderRadius: '18px', overflow: 'hidden', border: '1px solid var(--ifm-color-emphasis-300)', background: 'var(--ifm-background-surface-color)' };
  const header = { padding: '16px 24px', borderBottom: '1px solid var(--ifm-color-emphasis-200)' };
  const body = { padding: '24px' };
  const btnBase = { padding: '7px 18px', border: 'none', borderRadius: '8px', fontWeight: '700', fontSize: '13px', cursor: 'pointer', transition: 'all 0.2s' };

  return (
    <div style={wrap}>
      <div style={header}>
        <div style={{ fontWeight: '700', fontSize: '15px', marginBottom: '12px' }}>🔄 요청 처리 흐름 비교</div>
        <div style={{ display: 'flex', gap: '8px', flexWrap: 'wrap' }}>
          {['servlet', 'spring'].map(m => (
            <button key={m} onClick={() => setMode(m)} style={{ ...btnBase, background: mode === m ? (m === 'servlet' ? '#ef4444' : '#3b82f6') : 'var(--ifm-color-emphasis-200)', color: mode === m ? '#fff' : 'var(--ifm-color-content-secondary)' }}>
              {m === 'servlet' ? '서블릿 기반' : 'Spring MVC'}
            </button>
          ))}
        </div>
      </div>
      <div style={body}>
        <div style={{ display: 'flex', flexDirection: 'column', gap: '10px', marginBottom: '20px' }}>
          {steps.map((s, i) => (
            <div key={i} style={{ display: 'flex', alignItems: 'center', gap: '12px', padding: '12px 16px', borderRadius: '10px', border: `1px solid ${step >= i ? s.color + '44' : 'var(--ifm-color-emphasis-200)'}`, background: step >= i ? s.color + '11' : 'transparent', transition: 'all 0.4s', opacity: step >= i ? 1 : 0.4 }}>
              <div style={{ width: '28px', height: '28px', borderRadius: '50%', background: step >= i ? s.color : 'var(--ifm-color-emphasis-300)', color: '#fff', display: 'flex', alignItems: 'center', justifyContent: 'center', fontWeight: '700', fontSize: '12px', flexShrink: 0, transition: 'background 0.4s' }}>{i + 1}</div>
              <div>
                <div style={{ fontWeight: '700', fontSize: '13px', color: step >= i ? s.color : 'var(--ifm-color-content-secondary)' }}>{s.label}</div>
                {step >= i && <div style={{ fontSize: '12px', color: 'var(--ifm-color-content-secondary)', marginTop: '2px' }}>{s.desc}</div>}
              </div>
            </div>
          ))}
        </div>
        {mode === 'servlet' && step === steps.length - 1 && (
          <div style={{ padding: '12px 16px', borderRadius: '10px', background: 'rgba(239,68,68,0.08)', border: '1px solid rgba(239,68,68,0.3)', fontSize: '13px', color: '#ef4444', fontWeight: '600' }}>
            ⚠️ 기능 10개 → 서블릿 10개, 100개 → 100개. 규모가 커질수록 유지보수 지옥.
          </div>
        )}
        {mode === 'spring' && step === steps.length - 1 && (
          <div style={{ padding: '12px 16px', borderRadius: '10px', background: 'rgba(59,130,246,0.08)', border: '1px solid rgba(59,130,246,0.3)', fontSize: '13px', color: '#3b82f6', fontWeight: '600' }}>
            ✅ 진입점 하나, 공통 처리 자동화. 컨트롤러는 비즈니스 로직만 담당합니다.
          </div>
        )}
        <div style={{ display: 'flex', gap: '8px' }}>
          <button onClick={start} disabled={running} style={{ ...btnBase, background: running ? 'var(--ifm-color-emphasis-300)' : '#3b82f6', color: '#fff', opacity: running ? 0.6 : 1 }}>▶ 실행</button>
          <button onClick={reset} style={{ ...btnBase, background: 'var(--ifm-color-emphasis-200)', color: 'var(--ifm-color-content)' }}>↺ 초기화</button>
        </div>
      </div>
    </div>
  );
}

## 서블릿 기반 개발의 구조적 한계

초기 Java 웹 개발은 서블릿(Servlet) 클래스가 요청마다 1:1 대응하는 구조였습니다. 기능 수가 늘어날수록 클래스 수가 선형으로 증가하고, 공통 처리(형 변환, 뷰 이동, 예외 처리)가 모든 클래스에 반복되었습니다.

| 문제 | 구체적 부담 |
|---|---|
| 클래스 폭증 | 기능 N개 → 서블릿 클래스 N개 |
| 형 변환 반복 | 모든 파라미터가 `String`, 매번 수동 변환 |
| 뷰 경로 조합 | `"/WEB-INF/views/" + name + ".jsp"` 반복 |
| 공통 처리 분산 | 로그·인증 코드가 모든 서블릿에 흩어짐 |

이를 해결한 것이 **프론트 컨트롤러 패턴** — 단일 진입점이 모든 요청을 받아 "어디로 보낼지"를 결정하는 구조입니다. 스프링 MVC의 `DispatcherServlet`이 바로 이 역할입니다.

## Spring의 핵심 3축

스프링이 제공하는 기능은 수십 가지지만, **IoC/DI · AOP · MVC** 세 축이 나머지 모든 기능의 기반이 됩니다. 각각 클릭해서 확인해보세요.

<SpringCorePillars />

### 왜 이 3축인가?

- **IoC/DI** → 결합도를 낮추고 테스트·교체가 쉬워집니다.
- **AOP** → 공통 관심사를 분리해 핵심 로직이 순수해집니다.
- **MVC** → 요청~응답 파이프라인을 표준화해 레이어 책임이 명확해집니다.

세 가지가 함께 동작할 때 "SOLID 원칙을 자연스럽게 따르는 구조"가 됩니다.

## 빈(Bean)이란?

스프링 컨테이너가 **생성·관리하는 객체**를 빈이라고 합니다. `new`로 직접 만든 객체는 스프링이 모르므로 DI·AOP가 적용되지 않습니다.

```
일반 객체: new SamsungTv()  → 스프링이 모름, 주입·관리 불가
빈 객체: @Component SamsungTv → 컨테이너가 생성, 주입·관리 가능
```

### `@ComponentScan` 동작 원리

`@SpringBootApplication` 내부에는 `@ComponentScan`이 포함되어 있으며, **메인 클래스가 위치한 패키지와 그 하위 패키지**만 스캔합니다.

```
com.example.myapp
├── MyAppApplication.java  ← @SpringBootApplication (베이스 패키지)
├── user/
│   ├── UserController.java    ✅ 스캔됨
│   └── UserService.java       ✅ 스캔됨
└── util/
    └── DateUtils.java         ✅ 스캔됨

com.external.helper
└── HelperBean.java            ❌ 베이스 패키지 밖 → 스캔 안 됨
```

베이스 패키지 밖에 빈을 두면 주입 시 `NoSuchBeanDefinitionException`이 발생합니다. 가장 흔한 초기 오류 중 하나입니다.

## AOP 맛보기 — 공통 관심사 분리

```java
// Before AOP: 모든 서비스 메서드에 로그 코드가 흩어짐
public OrderDto createOrder(...) {
  log.info("createOrder 호출");  // 반복
  // 비즈니스 로직
  log.info("createOrder 완료");  // 반복
}

// After AOP: 어드바이스 하나로 일괄 처리
@Aspect @Component
public class LogAspect {
  @Around("execution(* com.example..*Service.*(..))")
  public Object logAround(ProceedingJoinPoint pjp) throws Throwable {
    log.info("호출: {}", pjp.getSignature());
    Object result = pjp.proceed();
    log.info("완료: {}", pjp.getSignature());
    return result;
  }
}
```

- **포인트컷**: "어디에 적용할지" — `execution(* com.example..*Service.*(..))`
- **어드바이스**: "무엇을 할지" — `@Around`, `@Before`, `@After` 등
- **어스펙트**: 포인트컷 + 어드바이스의 조합

`@Transactional`이 자동으로 트랜잭션을 열고 닫는 것도 AOP로 구현되어 있습니다.

## 정리 — Spring의 큰 그림

```
요청 → DispatcherServlet (프론트 컨트롤러)
           ↓ HandlerMapping → 컨트롤러 탐색
           ↓ HandlerAdapter → 파라미터 바인딩 + 컨트롤러 실행
           ↓ AOP 프록시 → 트랜잭션·로그 등 자동 처리
           ↓ IoC 컨테이너 → 빈 주입(DI)
           ↓ ViewResolver → 뷰 이름 → 실제 뷰 경로 조합
응답 ← 렌더링 결과 반환
```

| 개념 | 핵심 한 줄 |
|---|---|
| IoC / DI | 객체 생성·주입을 컨테이너에 위임한다 |
| AOP | 공통 관심사를 핵심 로직에서 분리한다 |
| Spring MVC | 단일 진입점(DispatcherServlet)으로 요청을 파이프라인으로 처리한다 |
| Bean | 컨테이너가 관리하는 객체 — `@Component` 또는 `@Bean`으로 등록 |
| Spring Boot | 자동 설정 + 스타터 + 내장 서버로 스프링 구성 부담을 줄인다 |

다음 글에서는 `@SpringBootApplication` 내부 구성, DI 주입 방식 3가지(`@Autowired` / `@Qualifier` / 생성자 주입) 실습을 상세히 다룹니다.
