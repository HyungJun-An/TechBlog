export function InjectionDecisionGuide() {
  const [answers, setAnswers] = useState({});
  const [result, setResult] = useState(null);

  const QUESTIONS = [
    { id: 'required', text: '의존 객체가 반드시 필요한가요?', yes: '필수', no: '선택적' },
    { id: 'multiple', text: '같은 타입 빈이 여러 개인가요?', yes: '다중 빈', no: '단일 빈' },
    { id: 'change', text: '생성 이후에도 의존성을 교체해야 하나요?', yes: '가변', no: '불변' },
  ];

  const RESULTS = {
    '필수-단일 빈-불변': { method: '생성자 주입', color: '#3b82f6', icon: '🏗️', note: '가장 일반적인 케이스. @RequiredArgsConstructor로 깔끔하게.' },
    '필수-다중 빈-불변': { method: '생성자 주입 + @Qualifier', color: '#3b82f6', icon: '🏗️', note: '파라미터에 @Qualifier("name")를 붙여 특정 빈을 주입.' },
    '필수-단일 빈-가변': { method: 'Setter 주입', color: '#f59e0b', icon: '🔧', note: '가변이 정말 필요한지 재검토 권장. 설계 냄새일 수 있음.' },
    '필수-다중 빈-가변': { method: 'Setter 주입 + @Qualifier', color: '#f59e0b', icon: '🔧', note: '가변+다중 조합. 설계를 다시 검토하는 것이 좋습니다.' },
    '선택적-단일 빈-불변': { method: 'Setter 주입 (required=false)', color: '#f59e0b', icon: '🔧', note: '@Autowired(required=false) 세터로 빈이 없어도 기동.' },
    '선택적-다중 빈-불변': { method: 'Setter 주입 + @Qualifier (required=false)', color: '#f59e0b', icon: '🔧', note: '빈이 없을 때 null 유지. 방어 로직 필요.' },
    '선택적-단일 빈-가변': { method: 'Setter 주입 (required=false)', color: '#f59e0b', icon: '🔧', note: '선택적이고 가변. 가장 유연하지만 가장 불안정한 조합.' },
    '선택적-다중 빈-가변': { method: 'Setter 주입 + @Qualifier (required=false)', color: '#f59e0b', icon: '🔧', note: '복잡도가 높습니다. 인터페이스 분리나 Optional 고려를 권장.' },
  };

  const toggle = (id, val) => {
    const next = { ...answers, [id]: val };
    setAnswers(next);
    if (Object.keys(next).length === QUESTIONS.length) {
      const key = `${next.required}-${next.multiple}-${next.change}`;
      setResult(RESULTS[key] || null);
    } else {
      setResult(null);
    }
  };

  const reset = () => { setAnswers({}); setResult(null); };
  const wrap = { margin: '32px 0', borderRadius: '18px', overflow: 'hidden', border: '1px solid var(--ifm-color-emphasis-300)', background: 'var(--ifm-background-surface-color)' };

  return (
    <div style={wrap}>
      <div style={{ padding: '16px 24px', borderBottom: '1px solid var(--ifm-color-emphasis-200)' }}>
        <div style={{ fontWeight: '700', fontSize: '15px', marginBottom: '4px' }}>🗺️ 어떤 주입 방식을 써야 할까?</div>
        <div style={{ fontSize: '12px', color: 'var(--ifm-color-content-secondary)' }}>질문에 답하면 적합한 주입 방식을 추천합니다</div>
      </div>
      <div style={{ padding: '24px' }}>
        <div style={{ display: 'flex', flexDirection: 'column', gap: '14px', marginBottom: '20px' }}>
          {QUESTIONS.map((q) => (
            <div key={q.id} style={{ padding: '14px 16px', borderRadius: '10px', background: 'var(--ifm-color-emphasis-100)', border: answers[q.id] ? '1px solid var(--ifm-color-primary)' : '1px solid transparent' }}>
              <div style={{ fontSize: '13px', fontWeight: '600', marginBottom: '10px', color: 'var(--ifm-color-content)' }}>{q.text}</div>
              <div style={{ display: 'flex', gap: '8px' }}>

`@Autowired`는 어디에 붙이느냐에 따라 동작 방식이 달라집니다. 필드에 붙이면 필드 주입, 생성자에 붙이면 생성자 주입, 세터에 붙이면 세터 주입입니다. 기능은 비슷해 보이지만 불변성·테스트성·오류 감지 시점에서 큰 차이가 있습니다.

## 주입 방식 4가지 한눈에

스프링은 `@Autowired`가 붙은 위치를 보고 빈을 주입합니다.

```
위치            방식
──────────────────────────────────────
필드 위        → 필드 주입 (Field Injection)
생성자 위      → 생성자 주입 (Constructor Injection)
세터 메서드 위 → Setter 주입 (Setter Injection)
일반 메서드 위 → 메서드 주입 (Method Injection)
```

탭으로 각 방식의 코드와 장단점을 비교해보세요.

## 생성자 주입이 권장되는 3가지 이유

스프링 공식 문서와 IntelliJ 모두 생성자 주입을 권장합니다. 단순히 "관례"가 아니라 구조적인 이유가 있습니다.

<ConstructorInjectionWhy />

### @RequiredArgsConstructor로 보일러플레이트 없애기

생성자 주입의 유일한 단점인 "생성자 코드가 길어짐"은 Lombok으로 해결합니다.

```java
// Before — 생성자 직접 작성
@Service
public class OrderService {
  private final PaymentClient paymentClient;
  private final NotifyService notifyService;
  private final OrderRepository orderRepository;

  public OrderService(PaymentClient paymentClient,
                      NotifyService notifyService,
                      OrderRepository orderRepository) {
    this.paymentClient = paymentClient;
    this.notifyService = notifyService;
    this.orderRepository = orderRepository;
  }
}

// After — @RequiredArgsConstructor
@Service
@RequiredArgsConstructor    // final 필드 생성자 자동 생성
public class OrderService {
  private final PaymentClient paymentClient;
  private final NotifyService notifyService;
  private final OrderRepository orderRepository;
  // 동일하게 동작, 코드 절반 이상 감소
}
```

## Setter 주입 — 선택적 의존성 처리

의존 객체가 없어도 애플리케이션이 동작해야 하는 경우에 사용합니다.

```java
@Service
public class ReportService {

  private NotifyService notifyService;   // 선택적 의존성

  // 빈이 있으면 주입, 없으면 이 메서드 자체가 호출되지 않음
  @Autowired(required = false)
  public void setNotifyService(NotifyService notifyService) {
    this.notifyService = notifyService;
  }

  public void generateReport() {
    // 방어 로직 필요
    if (notifyService != null) {
      notifyService.sendAlert("보고서 생성 완료");
    }
  }
}
```

> **언제 쓰나?** 플러그인 구조처럼 특정 모듈이 없어도 기본 동작은 해야 하는 경우. 레거시 시스템에서 특정 빈이 조건부로만 존재하는 경우.

---

## 메서드 주입 — Setter 주입의 확장

세터가 아닌 일반 메서드에 `@Autowired`를 붙여 여러 빈을 한 번에 주입받습니다.

```java
@Service
public class DashboardService {

  private MetricClient metricClient;
  private CacheClient cacheClient;
  private AlertService alertService;

  // 세터가 아닌 일반 메서드
  @Autowired
  public void setup(MetricClient metricClient,
                    CacheClient cacheClient,
                    AlertService alertService) {
    this.metricClient = metricClient;
    this.cacheClient = cacheClient;
    this.alertService = alertService;
  }
}
```

Setter 주입과 실질적으로 동일하며 파라미터를 여러 개 받을 수 있다는 차이만 있습니다. 생성자 주입으로 대체할 수 있어 실무에서는 거의 사용하지 않습니다.

## 주입 시 다중 빈 문제 — 빠른 복습

주입 방식과 무관하게, 같은 타입 빈이 2개 이상이면 추가 처리가 필요합니다.

```java
// 두 빈 등록
@Bean("acOn")  AirConditioner acOn()  { return new AirConditioner(true); }
@Bean("acOff") AirConditioner acOff() { return new AirConditioner(false); }

// ── 생성자 주입 + @Qualifier ──
public RoomService(@Qualifier("acOn") AirConditioner ac) { ... }

// ── 필드 주입 + @Qualifier (비권장) ──
@Autowired @Qualifier("acOn")
private AirConditioner ac;

// ── @Resource (이름 우선 탐색) ──
@Resource(name = "acOn")
private AirConditioner ac;
```

## 전체 정리

```
DI 주입 방식 4가지
├── 생성자 주입 ★★★★★  — final 불변 보장, 테스트 용이, 순환 의존성 조기 감지
├── Setter 주입 ★★☆☆☆  — 선택적 의존성 처리, required=false 조합
├── 메서드 주입 ★☆☆☆☆  — Setter의 확장형, 거의 사용 안 함
└── 필드 주입  ☆☆☆☆☆  — 프로덕션 비권장, IDE 경고, 테스트 어려움

결론:
├── 필수 의존성  → 생성자 주입 (@RequiredArgsConstructor로 간소화)
└── 선택적 의존성 → Setter 주입 (@Autowired(required=false))
```
