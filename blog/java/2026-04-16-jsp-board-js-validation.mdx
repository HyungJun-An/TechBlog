export function ReplyBoardDemo() {
  const initial = [
    { seq: 1, title: '공지사항: 게시판 이용 안내', id: 'admin', depth: 0, reRef: 1, delFlag: 'N' },
    { seq: 2, title: 'React vs JSP 어떤 게 나을까요?', id: 'user1', depth: 0, reRef: 2, delFlag: 'N' },
    { seq: 3, title: 'Re: React vs JSP 어떤 게 나을까요?', id: 'admin', depth: 1, reRef: 2, delFlag: 'N' },
    { seq: 4, title: 'Re: Re: 감사합니다! 많은 도움이 됐어요.', id: 'user1', depth: 2, reRef: 2, delFlag: 'N' },
    { seq: 5, title: 'JPA와 MyBatis 차이점 질문', id: 'user2', depth: 0, reRef: 5, delFlag: 'N' },
  ];
  const [posts, setPosts] = useState(initial);

  const toggleDel = (seq) => setPosts(prev => prev.map(p => p.seq === seq ? { ...p, delFlag: p.delFlag === 'N' ? 'Y' : 'N' } : p));
  const addDepth = (seq) => setPosts(prev => prev.map(p => p.seq === seq ? { ...p, depth: Math.min(p.depth + 1, 3) } : p));
  const subDepth = (seq) => setPosts(prev => prev.map(p => p.seq === seq ? { ...p, depth: Math.max(p.depth - 1, 0) } : p));
  const reset = () => setPosts(initial);

  const wrap = { margin: '32px 0', borderRadius: '18px', overflow: 'hidden', border: '1px solid var(--ifm-color-emphasis-300)', background: 'var(--ifm-background-surface-color)' };

  return (
    <div style={wrap}>
      <div style={{ padding: '16px 24px', borderBottom: '1px solid var(--ifm-color-emphasis-200)', display: 'flex', justifyContent: 'space-between', alignItems: 'center' }}>
        <div style={{ fontWeight: '700', fontSize: '15px' }}>📑 리스트 렌더링 시뮬레이터</div>
        <button onClick={reset} style={{ padding: '5px 12px', borderRadius: '8px', border: '1px solid var(--ifm-color-emphasis-300)', background: 'transparent', color: 'var(--ifm-color-content-secondary)', fontSize: '12px', cursor: 'pointer' }}>↺ 초기화</button>
      </div>

      <div style={{ overflowX: 'auto' }}>
        <table style={{ width: '100%', borderCollapse: 'collapse', fontSize: '13px' }}>
          <thead>
            <tr style={{ background: 'var(--ifm-color-emphasis-100)' }}>
              {['SEQ', '제목 (depth*20px 들여쓰기 적용)', 'depth', 'delFlag', '조작'].map((h, i) => (
                <th key={i} style={{ padding: '10px 14px', textAlign: i === 0 || i >= 2 ? 'center' : 'left', fontWeight: '700', color: 'var(--ifm-color-emphasis-600)', fontSize: '11px', whiteSpace: 'nowrap' }}>{h}</th>
              ))}
            </tr>
          </thead>
          <tbody>
            {posts.map((p, i) => (
              <tr key={p.seq} style={{ borderTop: '1px solid var(--ifm-color-emphasis-200)', background: i % 2 === 0 ? 'transparent' : 'var(--ifm-color-emphasis-50,rgba(0,0,0,0.02))' }}>
                <td style={{ padding: '10px 14px', textAlign: 'center', color: 'var(--ifm-color-emphasis-500)', fontFamily: 'monospace', fontSize: '12px' }}>{p.seq}</td>
                <td style={{ padding: '10px 14px' }}>
                  <div style={{ display: 'flex', alignItems: 'center', gap: '6px', paddingLeft: `${p.depth * 20}px`, transition: 'padding-left 0.2s' }}>
                    {p.depth > 0 && (
                      <span style={{ color: 'var(--ifm-color-emphasis-400)', fontSize: '13px', flexShrink: 0 }}>↳</span>
                    )}
                    {p.delFlag === 'Y' ? (
                      <span style={{ color: 'var(--ifm-color-emphasis-400)', fontStyle: 'italic', fontSize: '13px' }}>관리자에 의해 삭제되었습니다.</span>
                    ) : (
                      <span style={{ color: 'var(--ifm-color-primary)', fontWeight: '500', fontSize: '13px', cursor: 'pointer' }}>{p.title}</span>
                    )}
                  </div>
                </td>
                <td style={{ padding: '10px 14px', textAlign: 'center' }}>
                  <span style={{ padding: '2px 8px', borderRadius: '12px', background: p.depth > 0 ? '#8b5cf620' : 'var(--ifm-color-emphasis-100)', color: p.depth > 0 ? '#8b5cf6' : 'var(--ifm-color-emphasis-500)', fontSize: '12px', fontWeight: '700', fontFamily: 'monospace' }}>{p.depth}</span>
                </td>
                <td style={{ padding: '10px 14px', textAlign: 'center' }}>
                  <span style={{ padding: '2px 8px', borderRadius: '12px', background: p.delFlag === 'Y' ? '#ef444420' : '#10b98120', color: p.delFlag === 'Y' ? '#ef4444' : '#10b981', fontSize: '12px', fontWeight: '700', fontFamily: 'monospace' }}>{p.delFlag}</span>
                </td>
                <td style={{ padding: '8px 14px', textAlign: 'center' }}>
                  <div style={{ display: 'flex', gap: '4px', justifyContent: 'center' }}>
                    <button onClick={() => subDepth(p.seq)} title="depth 감소" style={{ padding: '3px 7px', borderRadius: '5px', border: '1px solid var(--ifm-color-emphasis-300)', background: 'transparent', color: 'var(--ifm-color-content-secondary)', fontSize: '11px', cursor: 'pointer' }}>d-</button>
                    <button onClick={() => addDepth(p.seq)} title="depth 증가" style={{ padding: '3px 7px', borderRadius: '5px', border: '1px solid var(--ifm-color-emphasis-300)', background: 'transparent', color: 'var(--ifm-color-content-secondary)', fontSize: '11px', cursor: 'pointer' }}>d+</button>

        d+ / d- 버튼으로 depth를 조절하고, 삭제 버튼으로 delFlag를 토글해보세요. JSP에서는 <code>c:forEach</code> + <code>c:if</code>로 이 로직을 구현합니다.
      </div>
    </div>
  );
}

  const [mode, setMode] = useState('raw');

  const escape = (str) =>
    str.replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;').replace(/"/g, '&quot;').replace(/'/g, '&#39;').replace(/\n/g, '<br>');

  const modes = [
    { key: 'raw',    label: '방어 없음',       color: '#ef4444', desc: 'JSP: ${dto.content}  — EL 직접 출력, HTML 태그 그대로 해석' },

    if (mode === 'raw') return { html: input, isDanger: true };

## MVC 전체 흐름 — 게시판 시나리오

> 사용자 요청 → 서블릿(컨트롤러) → DAO(DB) → 스코프 저장 → JSP 렌더링. 이 패턴이 리스트/작성/상세/수정/답글 전 시나리오를 관통한다.

컨트롤러는 URL 매핑(예: `boardList2`, `insertBoard2`)으로 분리되며, 역할은 단순하다: **파라미터 수신 → 비즈니스 로직(DAO) 호출 → 결과 전달(forward/redirect)**. JSP는 로직 없이 출력만 담당한다.

```java
// 상세 보기 — 전형적인 컨트롤러 패턴
int seq = Integer.parseInt(request.getParameter("seq"));
BoardDto dto = dao.getDetail(seq);     // DB 조회
dao.updateViewCnt(seq);                // 조회수 증가
request.setAttribute("dto", dto);     // Request 스코프 저장
RequestDispatcher rd = request.getRequestDispatcher("/WEB-INF/views/detailBoard.jsp");
rd.forward(request, response);
```

## 공통 레이아웃: 정적 include + Bootstrap

> JSP 정적 include는 컴파일 시점에 파일을 합치며, taglib 선언을 헤더에 한 번만 두면 모든 include 파일에서 JSTL을 사용할 수 있다.

`header.jsp`에 Bootstrap CSS 링크와 taglib 선언을 한 번 두면, include한 모든 JSP에서 재사용된다.

```jsp
<%-- header.jsp --%>
<%@ taglib prefix="c"   uri="http://java.sun.com/jsp/jstl/core" %>
<%@ taglib prefix="fn"  uri="http://java.sun.com/jsp/jstl/functions" %>
<%@ taglib prefix="fmt" uri="http://java.sun.com/jsp/jstl/fmt" %>
<link rel="stylesheet" href="https://cdn.../bootstrap.min.css">
```

```jsp
<%-- boardList.jsp — 각 화면에서 include --%>
<%@ include file="/WEB-INF/views/common/header.jsp" %>
...본문...
<%@ include file="/WEB-INF/views/common/footer.jsp" %>
```

리소스 파일(JS/CSS/이미지)은 반드시 `webapp/js/...` 위치에 두어야 한다. `WEB-INF` 하위에 두면 외부에서 직접 접근이 차단되어 정적 파일로 사용할 수 없다.

| 구분 | JSP 정적 include | React 컴포넌트 |
|---|---|---|
| 결합 시점 | 컴파일 시 합쳐짐 | 런타임 렌더링 |
| 데이터 전달 | request/session/EL로 암묵 공유 | props로 명시 전달 |
| taglib 전파 | 헤더 선언이 include 파일에 전파됨 | import/props 흐름 추적 필요 |

## XSS 방어

> `<script>` 태그를 그대로 저장하면 상세 페이지에서 브라우저가 실행한다. 입력·출력·서버 3단계 각각의 방어 전략이 있다.

```javascript
// JS 입력 단계 — 정규표현식 이스케이프
function escapeXss(str) {
  return str
    .replace(/&/g, '&amp;')
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;')
    .replace(/'/g, '&#39;');
}
```

```java
// 서버 전역 — Filter + RequestWrapper
public class XssFilter implements Filter {
  @Override
  public void doFilter(ServletRequest req, ServletResponse res, FilterChain chain)
          throws IOException, ServletException {
    chain.doFilter(new XssRequestWrapper((HttpServletRequest) req), res);
  }
}

public class XssRequestWrapper extends HttpServletRequestWrapper {
  @Override
  public String getParameter(String name) {
    String val = super.getParameter(name);
    return val == null ? null : val.replaceAll("<", "&lt;").replaceAll(">", "&gt;");
  }
}
```

## 폼 가드: 이탈 방지 + 유효성 검사

> `onbeforeunload`는 입력 감지 시 활성화하고, 정상 제출 직전 비활성화한다. 키 차단은 keyCode 기반 이벤트로 구현한다.

```javascript
let isShow = false;                // 이탈 경고 플래그

// 입력 감지 → 플래그 활성화
document.querySelector('#title').addEventListener('input', () => {
  isShow = true;
});

// 페이지 이탈 경고
window.onbeforeunload = function() {
  if (isShow) return '저장되지 않은 내용이 있습니다.';
};

// 제출 — 플래그 해제 후 전송
function submitForm() {
  const title = document.querySelector('#title').value.trim();
  const content = document.querySelector('#content').value.trim();
  if (!title) { alert('제목을 입력하세요.'); return; }
  if (!content) { alert('내용을 입력하세요.'); return; }
  isShow = false;           // 경고 없이 이탈 허용
  document.forms[0].submit();
}
```

```javascript
// F5/Ctrl+R/Ctrl+N 차단
let isCtrl = false;

document.addEventListener('keydown', function(e) {
  if (e.keyCode === 17) isCtrl = true;          // Ctrl 눌림 추적
  if (e.keyCode === 116 ||                       // F5
     (isCtrl && e.keyCode === 82) ||             // Ctrl+R
     (isCtrl && e.keyCode === 78)) {             // Ctrl+N
    e.preventDefault();
    e.stopPropagation();
    return false;
  }
});
document.addEventListener('keyup', function(e) {
  if (e.keyCode === 17) isCtrl = false;
});
```

## 정리

오늘 구현한 게시판의 공통 패턴은 다음 하나다.

```
요청 URL 매핑 (서블릿 doGet/doPost)
  → request.setCharacterEncoding("UTF-8")
  → 세션/파라미터 수신
  → DAO 실행 (SELECT/INSERT/UPDATE)
  → 성공: setAttribute → forward / sendRedirect
  → 실패: alert 스크립트 또는 error.jsp forward
```

| 계층 | 핵심 구현 | 주의점 |
|---|---|---|
| 컨트롤러 | URL 매핑, 파라미터 수신, DAO 호출 | 인코딩 설정 항상 첫 번째 |
| JSTL/EL | `c:if`, `c:forEach`, `fn:length`, `fmt:formatDate` | JSTL 주석 내 코드 실행 주의 |
| JS | onbeforeunload, keyCode 차단, 유효성 검사 | 브라우저 UI 버튼 새로고침은 완전 차단 불가 |
| XSS | c:out, 정규표현식, Filter+Wrapper | 수정 화면에도 동일 방어 필요 |

세션 체크, XSS 처리, 인코딩 설정처럼 서블릿마다 반복되는 코드는 모두 **Filter**로 공통화할 수 있다. 지금은 동작 확인이 목적이었고, 이후 리팩토링 대상이다.
