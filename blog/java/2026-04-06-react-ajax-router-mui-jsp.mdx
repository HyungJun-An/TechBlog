import { useState, useEffect, useRef } from 'react';

export function FetchFlowViz() {
  const STEPS = [
    { id: 0, label: '① 폼 입력',         sub: 'firstName, lastName, email',              icon: '📝', color: '#3b82f6' },
    { id: 1, label: '② fetch 호출',       sub: 'method: POST, headers 설정',             icon: '📡', color: '#8b5cf6' },
    { id: 2, label: '③ JSON 직렬화',      sub: 'JSON.stringify(body)',                   icon: '🔄', color: '#f59e0b' },
    { id: 3, label: '④ HTTP 요청 전송',   sub: 'Content-Type: application/json',         icon: '🚀', color: '#06b6d4' },
    { id: 4, label: '⑤ 스프링 수신',      sub: '@RequestBody FormRequest req',           icon: '🌿', color: '#10b981' },
    { id: 5, label: '⑥ JSON 응답',        sub: 'ResponseEntity.ok(response)',            icon: '📦', color: '#10b981' },
    { id: 6, label: '⑦ res.json() 파싱', sub: '.then(res => res.json())',               icon: '✅', color: '#3b82f6' },
  ];

  const [active, setActive] = useState(-1);
  const [running, setRunning] = useState(false);
  const timerRef = useRef(null);

  const run = () => {
    setActive(-1);
    setRunning(true);
    let i = 0;
    const next = () => {
      setActive(i);
      i++;
      if (i < STEPS.length) timerRef.current = setTimeout(next, 700);
      else setRunning(false);
    };
    timerRef.current = setTimeout(next, 300);
  };

  const reset = () => {
    clearTimeout(timerRef.current);
    setActive(-1);
    setRunning(false);
  };

  useEffect(() => () => clearTimeout(timerRef.current), []);

  const wrap = { margin: '32px 0', borderRadius: '18px', overflow: 'hidden', border: '1px solid var(--ifm-color-emphasis-300)', background: 'var(--ifm-background-surface-color)' };
  const header = { padding: '18px 24px', borderBottom: '1px solid var(--ifm-color-emphasis-200)', display: 'flex', justifyContent: 'space-between', alignItems: 'center' };

  return (
    <div style={wrap}>
      <div style={header}>
        <div style={{ fontWeight: '700', fontSize: '15px' }}>🔁 fetch POST 전체 흐름</div>
        <div style={{ display: 'flex', gap: '8px' }}>
          <button onClick={run} disabled={running} style={{ padding: '7px 16px', borderRadius: '8px', border: 'none', background: running ? 'var(--ifm-color-emphasis-300)' : 'var(--ifm-color-primary)', color: running ? 'var(--ifm-color-content-secondary)' : '#fff', fontWeight: '600', fontSize: '13px', cursor: running ? 'not-allowed' : 'pointer' }}>
            {running ? '진행 중...' : '▶ 실행'}
          </button>
          <button onClick={reset} style={{ padding: '7px 12px', borderRadius: '8px', border: '1px solid var(--ifm-color-emphasis-300)', background: 'transparent', color: 'var(--ifm-color-content-secondary)', fontSize: '13px', cursor: 'pointer' }}>↺</button>
        </div>
      </div>
      <div style={{ padding: '24px', display: 'flex', flexDirection: 'column', gap: '10px' }}>
        {STEPS.map((step, i) => {
          const isPast   = i < active;
          const isCurrent = i === active;
          const opacity  = active === -1 ? 0.5 : (isPast || isCurrent ? 1 : 0.3);
          const bg       = isCurrent ? `${step.color}18` : isPast ? 'rgba(16,185,129,0.06)' : 'var(--ifm-color-emphasis-100)';
          const border   = isCurrent ? `1.5px solid ${step.color}` : isPast ? '1.5px solid rgba(16,185,129,0.35)' : '1.5px solid var(--ifm-color-emphasis-200)';
          return (
            <div key={i} style={{ display: 'flex', alignItems: 'center', gap: '14px', padding: '12px 16px', borderRadius: '10px', background: bg, border, opacity, transition: 'all 0.3s ease' }}>
              <span style={{ fontSize: '20px', minWidth: '28px', textAlign: 'center' }}>{isPast ? '✅' : step.icon}</span>
              <div style={{ flex: 1 }}>
                <div style={{ fontWeight: '600', fontSize: '13px', color: isCurrent ? step.color : isPast ? '#10b981' : 'var(--ifm-color-content)' }}>{step.label}</div>
                <div style={{ fontSize: '11px', color: 'var(--ifm-color-emphasis-500)', marginTop: '2px', fontFamily: 'monospace' }}>{step.sub}</div>
              </div>
              {isCurrent && (
                <div style={{ width: '8px', height: '8px', borderRadius: '50%', background: step.color, boxShadow: `0 0 8px ${step.color}` }} />
              )}
            </div>
          );
        })}
      </div>
      {active === -1 && (
        <div style={{ padding: '14px 24px', borderTop: '1px solid var(--ifm-color-emphasis-200)', fontSize: '13px', color: 'var(--ifm-color-emphasis-400)', textAlign: 'center' }}>
          ▶ 실행을 눌러 리액트 → 스프링부트 POST 흐름을 단계별로 확인하세요
        </div>
      )}
    </div>
  );
}

          <div style={{ marginBottom: '12px', padding: '12px 14px', borderRadius: '8px', background: 'rgba(16,185,129,0.08)', border: '1px solid rgba(16,185,129,0.3)', fontSize: '13px', color: '#10b981', fontWeight: '600' }}>

export function RouterViz() {
  const [mode, setMode]   = useState('spa');
  const [page, setPage]   = useState('home');
  const [loading, setLoading] = useState(false);
  const [reloadFlash, setReloadFlash] = useState(false);
  const [navCount, setNavCount] = useState(0);

  const PAGES = {
    home:      { label: 'Home',    icon: '🏠', color: '#3b82f6', content: 'Welcome! 메인 화면입니다.' },
    about:     { label: 'About',   icon: '👤', color: '#8b5cf6', content: '소개 페이지입니다.' },
    container: { label: 'Items',   icon: '📦', color: '#f59e0b', content: '상품 목록 페이지입니다.' },
  };

  const navigate = (to) => {
    if (to === page) return;
    setNavCount(c => c + 1);
    if (mode === 'reload') {
      setLoading(true);
      setReloadFlash(true);
      setTimeout(() => setReloadFlash(false), 150);
      setTimeout(() => { setLoading(false); setPage(to); }, 900);
    } else {
      setPage(to);
    }
  };

  const wrap = { margin: '32px 0', borderRadius: '18px', overflow: 'hidden', border: '1px solid var(--ifm-color-emphasis-300)', background: 'var(--ifm-background-surface-color)' };

    padding: '7px 14px', borderRadius: '7px', border: 'none', cursor: 'pointer', fontSize: '13px', fontWeight: page === key ? '600' : '400',
    background: page === key ? `${PAGES[key].color}18` : 'transparent',
    color: page === key ? PAGES[key].color : 'var(--ifm-color-emphasis-600)',
  });

  return (
    <div style={wrap}>
      <div style={header}>
        <div style={{ fontWeight: '700', fontSize: '15px' }}>🗺️ SPA 라우팅 vs 전통 방식</div>

          {[['spa', 'SPA (Link)'], ['reload', '전통 방식']].map(([val, label]) => (

'var(--ifm-color-primary)' : 'transparent', color: mode === val ? '#fff' : 'var(--ifm-color-content-secondary)', fontWeight: '600', fontSize: '12px', cursor: 'pointer' }}>
              {label}
            </button>
          ))}
        </div>
      </div>

      <div style={{ padding: '20px 24px' }}>
        <div style={{ padding: '10px 12px', background: 'var(--ifm-color-emphasis-100)', borderRadius: '10px', marginBottom: '16px', display: 'flex', gap: '8px', alignItems: 'center' }}>
          <span style={{ fontSize: '11px', color: 'var(--ifm-color-emphasis-500)', fontFamily: 'monospace' }}>🔗 localhost:5173/{page === 'home' ? '' : page}</span>
        </div>

        <div style={{ display: 'flex', gap: '8px', marginBottom: '16px' }}>
          {Object.entries(PAGES).map(([key, val]) => (
            <button key={key} onClick={() => navigate(key)} disabled={loading}
              style={{ ...navItem(key), transition: 'all 0.2s' }}>
              {val.icon} {val.label}

          ))}

          {loading

                <div style={{ fontWeight: '600', color: PAGES[page].color }}>{PAGES[page].label}</div>

        <div style={{ marginTop: '14px', display: 'flex', gap: '10px' }}>
          <div style={{ flex: 1, padding: '10px 14px', borderRadius: '8px', background: mode === 'spa' ? 'rgba(16,185,129,0.08)' : 'var(--ifm-color-emphasis-100)', border: mode === 'spa' ? '1px solid rgba(16,185,129,0.3)' : '1px solid var(--ifm-color-emphasis-200)', fontSize: '12px' }}>
            <div style={{ fontWeight: '700', color: '#10b981', marginBottom: '4px' }}>SPA — React Router</div>
            <div style={{ color: 'var(--ifm-color-emphasis-600)' }}>HTML/JS 재다운로드 없이 컴포넌트만 교체. 이동 횟수: <strong>{mode === 'spa' ? navCount : 0}</strong></div>
          </div>
          <div style={{ flex: 1, padding: '10px 14px', borderRadius: '8px', background: mode === 'reload' ? 'rgba(239,68,68,0.07)' : 'var(--ifm-color-emphasis-100)', border: mode === 'reload' ? '1px solid rgba(239,68,68,0.2)' : '1px solid var(--ifm-color-emphasis-200)', fontSize: '12px' }}>

            <div style={{ color: 'var(--ifm-color-emphasis-600)' }}>페이지마다 HTML 전체 재요청. 이동 횟수: <strong>{mode === 'reload' ? navCount : 0}</strong></div>
          </div>
        </div>
      </div>
    </div>
  );
}

export function MUIListDemo() {
  const [items, setItems]     = useState([{ product: '아메리카노', amount: '2' }, { product: '라떼', amount: '1' }]);
  const [open, setOpen]       = useState(false);
  const [form, setForm]       = useState({ product: '', amount: '' });
  const [removed, setRemoved] = useState(null);

  const handleChange = (e) => setForm({ ...form, [e.target.name]: e.target.value });

  const addItem = () => {
    if (!form.product || !form.amount) return;
    setItems(prev => [...prev, { ...form }]);
    setOpen(false);
    setForm({ product: '', amount: '' });
  };

  const removeItem = (idx) => {
    setRemoved(idx);
    setTimeout(() => {
      setItems(prev => prev.filter((_, i) => i !== idx));
      setRemoved(null);
    }, 300);
  };

  const wrap = { margin: '32px 0', borderRadius: '18px', overflow: 'hidden', border: '1px solid var(--ifm-color-emphasis-300)', background: 'var(--ifm-background-surface-color)' };
  const header = { padding: '16px 20px', background: '#1976d2', display: 'flex', justifyContent: 'space-between', alignItems: 'center' };

  return (
    <div style={wrap}>
      <div style={header}>
        <span style={{ color: '#fff', fontWeight: '700', fontSize: '15px' }}>🛒 MUI 리스트 추가/삭제 시뮬레이션</span>
        <button onClick={() => setOpen(true)} style={{ padding: '6px 14px', borderRadius: '6px', border: 'none', background: '#fff', color: '#1976d2', fontWeight: '700', fontSize: '13px', cursor: 'pointer' }}>+ 추가</button>
      </div>

## 3. OpenWeatherMap API — 날씨 데이터 출력

**외부 API 연동 절차: 가입 → API Key 발급 → DOC 확인 → 엔드포인트 호출.**

응답 JSON은 중첩 구조이므로 필요한 값의 경로를 파악하는 것이 핵심입니다.

| 상태(State) | 응답 경로 | 용도 |
|---|---|---|
| `temp` | `data.main.temp` | 현재 기온 표시 |
| `desc` | `data.weather[0].description` | 날씨 텍스트 |
| `icon` | `data.weather[0].icon` | 아이콘 URL 조합 (`50n` → `50n@2x.png`) |
| `isReady` | — | 로딩 완료 플래그, 마지막에 `true`로 전환 |

```jsx
useEffect(() => {
  fetch(`https://api.openweathermap.org/data/2.5/weather?q=Seoul&appid=${KEY}&units=metric`)
    .then((res) => res.json())
    .then((data) => {
      setTemp(data.main.temp);
      setDesc(data.weather[0].description);
      setIcon(data.weather[0].icon);
      setReady(true);     // 모든 값 세팅 완료 후 플래그 전환
    });
}, []);

return isReady
  ? <img src={`https://openweathermap.org/img/wn/${icon}@2x.png`} />
  : <p>날씨 로딩 중...</p>;
```

> **`isReady`를 마지막에 세팅하는 이유** — temp만 먼저 세팅되고 icon이 아직 빈 값이면 깨진 이미지가 순간 노출됩니다. 모든 상태가 준비된 뒤 한 번에 전환합니다.

`@CrossOrigin`은 빠른 해결책이지만 개발 단계용입니다. **전체 엔드포인트에 CORS를 적용하려면** `WebMvcConfigurer.addCorsMappings`를 사용합니다.

// 전역 CORS 설정 (운영 환경 권장)

    public void addCorsMappings(CorsRegistry registry) {
        registry.addMapping("/api/**")
                .allowedOrigins("http://localhost:5173")
                .allowedMethods("GET", "POST", "PUT", "DELETE");
    }
}

## 8. JSP/서블렛과 Jsoup 스크래핑

**Jsoup은 URL로 원격 HTML을 가져와 CSS 선택자로 DOM을 탐색하는 Java 라이브러리입니다.**

아래 파이프라인을 실행해 서블렛 → Jsoup → JSP 전체 흐름을 단계별로 확인하세요.

<ScrapingFlow />

```java
// 서블렛 핵심 코드
String url    = "https://comic.naver.com/webtoon/detail?titleId=" + titleId;
Document doc  = Jsoup.connect(url).get();                    // HTML 전체 문서
Elements imgs = doc.select("#comic_view_area img");          // CSS 선택자로 탐색

List<String> srcList = new ArrayList<>();
for (Element img : imgs) srcList.add(img.attr("src"));       // src 속성 수집

request.setAttribute("images", srcList);
request.getRequestDispatcher("/webtoon.jsp").forward(request, response);
```

```jsp
<%-- webtoon.jsp --%>
<% List<String> images = (List<String>) request.getAttribute("images");
   for (String src : images) { %>
  <img src="<%= src %>" style="width:100%;" />
<% } %>
```

> **법적 주의** — 스크래핑 대상의 이용약관과 저작권을 반드시 확인하세요. 이미지·콘텐츠를 상업적으로 재배포하면 법적 문제가 발생합니다.

## 9. WS vs WAS vs JSP, 그리고 SPA

| 방식 | HTML 생성 주체 | 데이터 형태 | 특징 |
|---|---|---|---|
| JSP / Thymeleaf | 서버(WAS) | 완성된 HTML | 서버 부하, SEO 유리 |
| React (SPA) | 브라우저 | JSON → 컴포넌트 렌더링 | 인터랙션 빠름, 초기 로딩 최적화 필요 |

최종 브라우저가 받는 것은 둘 다 HTML이지만 **HTML을 조립하는 위치**가 다릅니다.

## 선택 기준 요약

| 상황 | 선택 |
|---|---|

| CORS (전역 설정) | `WebMvcConfigurer.addCorsMappings` |
