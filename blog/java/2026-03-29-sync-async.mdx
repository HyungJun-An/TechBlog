import { useState, useEffect, useRef } from 'react';

export function SyncAsyncCompare() {
  const SPEED = 3;
  const TASKS = [
    { name: 'DB 조회',   simMs: 1200 },
    { name: 'API 호출',  simMs: 1800 },
    { name: '파일 읽기', simMs:  900 },
  ];
  const BAR_COLORS = ['#60a5fa', '#fb923c', '#34d399'];
  const totalSync  = TASKS.reduce((s, t) => s + t.simMs, 0);
  const totalAsync = Math.max(...TASKS.map(t => t.simMs));

  const [phase,      setPhase]      = useState('idle');
  const [syncProg,   setSyncProg]   = useState([0, 0, 0]);
  const [asyncProg,  setAsyncProg]  = useState([0, 0, 0]);
  const [syncMs,     setSyncMs]     = useState(0);
  const [asyncMs,    setAsyncMs]    = useState(0);
  const [syncFinal,  setSyncFinal]  = useState(null);
  const [asyncFinal, setAsyncFinal] = useState(null);
  const [syncCurIdx, setSyncCurIdx] = useState(0);

  const rafRef      = useRef(null);
  const startRef    = useRef(0);
  const syncIdxRef  = useRef(0);
  const syncStepRef = useRef(0);
  const sDoneRef    = useRef(false);
  const aDoneRef    = useRef(false);

  const go = () => {
    if (rafRef.current) cancelAnimationFrame(rafRef.current);
    startRef.current = 0; syncIdxRef.current = 0; syncStepRef.current = 0;
    sDoneRef.current = false; aDoneRef.current = false;
    setSyncProg([0,0,0]); setAsyncProg([0,0,0]);
    setSyncMs(0); setAsyncMs(0);
    setSyncFinal(null); setAsyncFinal(null);
    setSyncCurIdx(0); setPhase('running');

    const tick = (now) => {
      if (startRef.current === 0) { startRef.current = now; syncStepRef.current = now; }
      const sim = (now - startRef.current) * SPEED;

      if (!aDoneRef.current) {
        const ap = TASKS.map(t => Math.min(sim / t.simMs, 1));
        setAsyncProg(ap); setAsyncMs(sim);
        if (ap.every(p => p >= 1)) { aDoneRef.current = true; setAsyncFinal(sim); }
      }

      if (!sDoneRef.current) {
        const idx = syncIdxRef.current;
        if (idx < TASKS.length) {
          const p = Math.min((now - syncStepRef.current) * SPEED / TASKS[idx].simMs, 1);
          setSyncProg(prev => { const n = [...prev]; n[idx] = p; return n; });
          setSyncMs(sim);
          if (p >= 1) {
            const next = idx + 1; syncIdxRef.current = next; syncStepRef.current = now; setSyncCurIdx(next);
            if (next >= TASKS.length) { sDoneRef.current = true; setSyncFinal(sim); }
          }
        }
      }

      if (sDoneRef.current && aDoneRef.current) { setPhase('done'); return; }
      rafRef.current = requestAnimationFrame(tick);
    };
    rafRef.current = requestAnimationFrame(tick);
  };

  const reset = () => {
    if (rafRef.current) cancelAnimationFrame(rafRef.current);
    setPhase('idle'); setSyncProg([0,0,0]); setAsyncProg([0,0,0]);
    setSyncMs(0); setAsyncMs(0); setSyncFinal(null); setAsyncFinal(null);
    setSyncCurIdx(0); startRef.current = 0;
  };

  useEffect(() => () => { if (rafRef.current) cancelAnimationFrame(rafRef.current); }, []);

  const wrap = { margin: '32px 0', borderRadius: '18px', overflow: 'hidden', border: '1px solid var(--ifm-color-emphasis-300)', background: 'var(--ifm-background-surface-color)' };
  const header = { padding: '20px 24px', borderBottom: '1px solid var(--ifm-color-emphasis-200)', display: 'flex', justifyContent: 'space-between', alignItems: 'center' };
  const body = { padding: '24px', display: 'flex', gap: '16px', flexWrap: 'wrap' };
  const syncCard = { flex: '1 1 240px', background: 'rgba(59,130,246,0.06)', border: '1px solid rgba(59,130,246,0.2)', borderRadius: '12px', padding: '18px' };
  const asyncCard = { flex: '1 1 240px', background: 'rgba(249,115,22,0.06)', border: '1px solid rgba(249,115,22,0.2)', borderRadius: '12px', padding: '18px' };

  const renderPanel = (prog, ms, final, curIdx, isSync) => {
    const accent = isSync ? '#3b82f6' : '#f97316';
    const label  = isSync ? '동기 (Sync)' : '비동기 (Async)';
    const note   = isSync ? `순차 처리 · 예상 ${(totalSync/1000).toFixed(1)}s` : `병렬 처리 · 예상 ${(totalAsync/1000).toFixed(1)}s`;
    return (
      <div style={isSync ? syncCard : asyncCard}>
        <div style={{ marginBottom: '18px' }}>
          <div style={{ fontWeight: '700', fontSize: '14px', color: accent }}>{label}</div>
          <div style={{ fontSize: '12px', color: 'var(--ifm-color-emphasis-500)', marginTop: '2px' }}>{note}</div>
        </div>
        {TASKS.map((task, i) => {
          const p = prog[i];
          const done = p >= 1;
          const active = isSync ? (i === curIdx && phase === 'running') : (phase === 'running' && p < 1);
          const waiting = isSync && phase === 'running' && i > curIdx && p === 0;
          const barColor = isSync ? '#3b82f6' : BAR_COLORS[i];
          return (
            <div key={i} style={{ marginBottom: '14px' }}>
              <div style={{ display: 'flex', justifyContent: 'space-between', marginBottom: '5px', fontSize: '13px' }}>
                <span style={{ color: 'var(--ifm-color-content)', display: 'flex', alignItems: 'center', gap: '6px' }}>
                  {task.name}
                  {active && !done && <span style={{ width: '6px', height: '6px', borderRadius: '50%', background: barColor, display: 'inline-block' }} />}
                  {done && <span style={{ color: '#10b981', fontSize: '11px' }}>✓</span>}
                  {waiting && <span style={{ color: 'var(--ifm-color-emphasis-400)', fontSize: '11px' }}>대기</span>}
                </span>
                <span style={{ fontSize: '12px', color: 'var(--ifm-color-emphasis-500)', fontVariantNumeric: 'tabular-nums' }}>{task.simMs}ms</span>
              </div>
              <div style={{ height: '8px', background: 'var(--ifm-color-emphasis-200)', borderRadius: '4px', overflow: 'hidden' }}>
                <div style={{ height: '100%', width: `${p * 100}%`, background: done ? '#10b981' : barColor, borderRadius: '4px', transition: 'width 0.03s linear', boxShadow: active && !done ? `0 0 6px ${barColor}80` : 'none' }} />
              </div>
            </div>
          );
        })}
        <div style={{ marginTop: '18px', padding: '12px 14px', background: 'var(--ifm-color-emphasis-100)', borderRadius: '10px' }}>
          <div style={{ fontSize: '11px', color: 'var(--ifm-color-emphasis-500)', marginBottom: '3px', letterSpacing: '0.04em' }}>
            {final ? '완료' : phase === 'running' ? '경과 시간' : '예상 시간'}
          </div>
          <div style={{ fontSize: '32px', fontWeight: '700', fontFamily: 'monospace', fontVariantNumeric: 'tabular-nums', color: final ? '#10b981' : 'var(--ifm-color-content)', lineHeight: 1.1 }}>
            {final ? `${(final/1000).toFixed(2)}s` : phase === 'running' ? `${(ms/1000).toFixed(2)}s` : isSync ? `~${(totalSync/1000).toFixed(1)}s` : `~${(totalAsync/1000).toFixed(1)}s`}
          </div>
        </div>
      </div>
    );
  };

  return (
    <div style={wrap}>
      <div style={header}>
        <div style={{ fontWeight: '700', fontSize: '15px' }}>⚡ 실시간 실행 비교</div>
        <div style={{ display: 'flex', gap: '8px' }}>
          <button onClick={go} disabled={phase === 'running'} style={{ padding: '8px 18px', borderRadius: '8px', border: 'none', background: phase === 'running' ? 'var(--ifm-color-emphasis-300)' : 'var(--ifm-color-primary)', color: phase === 'running' ? 'var(--ifm-color-content-secondary)' : '#fff', fontWeight: '600', fontSize: '13px', cursor: phase === 'running' ? 'not-allowed' : 'pointer' }}>
            {phase === 'running' ? '실행 중...' : '▶ 동시 실행'}
          </button>
          <button onClick={reset} style={{ padding: '8px 14px', borderRadius: '8px', border: '1px solid var(--ifm-color-emphasis-300)', background: 'transparent', color: 'var(--ifm-color-content-secondary)', fontSize: '13px', cursor: 'pointer' }}>↺</button>
        </div>
      </div>
      <div style={body}>
        {renderPanel(syncProg,  syncMs,  syncFinal,  syncCurIdx, true)}
        {renderPanel(asyncProg, asyncMs, asyncFinal, 0,          false)}
      </div>
      {phase === 'done' && syncFinal && asyncFinal && (
        <div style={{ margin: '0 24px 24px', padding: '14px 18px', borderRadius: '10px', background: 'rgba(16,185,129,0.08)', border: '1px solid rgba(16,185,129,0.2)', display: 'flex', alignItems: 'center', gap: '14px' }}>
          <span style={{ fontSize: '22px' }}>🚀</span>
          <div>
            <div style={{ fontWeight: '700', fontSize: '14px', color: '#10b981' }}>비동기가 {(syncFinal / asyncFinal).toFixed(1)}배 더 빠릅니다</div>
            <div style={{ fontSize: '12px', color: 'var(--ifm-color-emphasis-500)', marginTop: '3px' }}>동기: {(syncFinal/1000).toFixed(2)}s · 비동기: {(asyncFinal/1000).toFixed(2)}s · 절약: {((syncFinal-asyncFinal)/1000).toFixed(2)}s</div>
          </div>
        </div>
      )}
      {phase === 'idle' && (
        <div style={{ padding: '16px 24px', borderTop: '1px solid var(--ifm-color-emphasis-200)', fontSize: '13px', color: 'var(--ifm-color-emphasis-400)', textAlign: 'center' }}>
          두 방식이 동시에 시작됩니다. 차이를 직접 눈으로 확인해보세요 →
        </div>
      )}
    </div>
  );
}

## 3. JavaScript의 비동기 처리

### Event Loop — 싱글 스레드가 비동기를 다루는 방법

JavaScript는 **싱글 스레드**입니다. 그럼에도 수천 개의 동시 요청을 처리할 수 있는 이유는 **Event Loop** 덕분입니다.

```
┌─────────────────────────────────────────────────────┐
│                  JavaScript Runtime                  │
│                                                     │
│  ┌──────────────┐     ┌───────────────────────────┐ │
│  │  Call Stack  │     │    Web API (libuv / OS)   │ │
│  │              │     │                           │ │
│  │  render()    │────▶│  fetch('/api')  ···진행중  │ │
│  │  main()      │     │  setTimeout()  ···진행중  │ │
│  └──────┬───────┘     └─────────────┬─────────────┘ │
│         │  비었을 때만               │ 완료 시        │
│         │  꺼냄                      ▼               │
│         │             ┌─────────────────────────┐   │
│         └────────────▶│      Task Queue          │   │
│           Event Loop  │  callback1, callback2…  │   │
│                        └─────────────────────────┘   │
└─────────────────────────────────────────────────────┘
```

- `fetch()`를 호출하면 Call Stack에서 바로 제거되고 OS에 I/O 위임
- Call Stack이 빈 상태일 때만 Event Loop이 Task Queue에서 콜백을 꺼냄
- 즉, **I/O 대기 중에도 Call Stack은 다른 코드를 실행**할 수 있음

> ⚠️ `while(true)` 같은 CPU 연산이 오래 걸리면 Call Stack이 막혀 Event Loop 전체가 멈춥니다.

## 4. Java의 비동기 처리

### Thread 모델 — 멀티 스레드가 비동기를 다루는 방법

Java는 기본적으로 **멀티 스레드** 모델입니다. 요청마다 별도 스레드를 할당하고, 각 스레드가 독립적으로 I/O를 처리합니다.

```
┌─────────────────────────────────────────────────────┐
│                   Java Runtime (JVM)                 │
│                                                     │
│  요청 A ──▶  Thread #1 ──▶ DB 쿼리 (블로킹 대기)    │
│  요청 B ──▶  Thread #2 ──▶ API 호출 (블로킹 대기)   │
│  요청 C ──▶  Thread #3 ──▶ 파일 읽기 (블로킹 대기)  │
│                                                     │
│  각 스레드: ~1MB 스택 메모리 점유                    │
│  10,000 동시 요청 = 10,000 스레드 = ~10GB           │
└─────────────────────────────────────────────────────┘
```

스레드가 I/O를 기다리는 동안 **OS 스레드를 점유**합니다. 동시 접속이 많아질수록 메모리와 컨텍스트 스위칭 비용이 급증합니다. 이것이 Java가 전통적으로 비동기 처리를 도입한 이유입니다.

## 5. Java vs JavaScript — 핵심 차이

<CompareTable />

### 가장 중요한 차이: 동시성 단위

- **JavaScript** — 하나의 스레드가 Event Loop으로 **제어권을 번갈아가며** 처리. `await`는 실제로 스레드를 멈추는 게 아니라 **함수 실행을 일시 중단**하고 제어권을 Event Loop에 반환.

- **Java** — 각 작업이 **실제로 다른 OS 스레드에서 병렬 실행**. `CompletableFuture.supplyAsync()`는 별도 스레드에서 진짜 동시에 돌아감.

```javascript
// JavaScript: 싱글 스레드 내 제어권 전환
async function main() {
  const a = await fetchA(); // 스레드를 멈추는 게 아님
  const b = await fetchB(); // fetchA 완료 후 실행 (직렬!)
}

// 진짜 병렬을 원하면 Promise.all
const [a, b] = await Promise.all([fetchA(), fetchB()]);
```

```java
// Java: 실제로 다른 스레드에서 병렬 실행
CompletableFuture<String> fa = CompletableFuture.supplyAsync(() -> fetchA());
CompletableFuture<String> fb = CompletableFuture.supplyAsync(() -> fetchB());
// fa와 fb는 서로 다른 스레드에서 실제 동시 실행 중
CompletableFuture.allOf(fa, fb).thenRun(() -> merge(fa.join(), fb.join()));
```

### Java 21 Virtual Threads — 게임 체인저

Java 21의 **Virtual Threads**는 두 세계의 장점을 결합합니다.

- OS 스레드 1개가 수천 개의 가상 스레드를 관리
- 블로킹 I/O 호출 시 OS 스레드를 반납, 다른 가상 스레드에 양보
- **기존 블로킹 코드를 그대로 사용**하면서 Node.js 수준의 동시성 달성

```java
// 기존 블로킹 코드 그대로 — Virtual Thread가 알아서 처리
try (var pool = Executors.newVirtualThreadPerTaskExecutor()) {
    for (int i = 0; i < 100_000; i++) {
        pool.submit(() -> {
            String data = jdbc.query("SELECT ...");  // 블로킹이지만 OK
            process(data);
        });
    }
}
```

## 6. 실무 선택 가이드

| 상황 | 선택 | 이유 |
| :--- | :--- | :--- |
| 독립적인 I/O 작업 여럿 | `CompletableFuture.allOf()` | 병렬 처리, 총 시간 = max(각 작업) |
| 이전 결과가 다음 입력 | 동기 또는 `.thenApply()` 체인 | 순서 의존성, 병렬화 불가 |
| Spring 메서드 단위 비동기화 | `@Async` + `@EnableAsync` | 선언형, 간결 |
| 대규모 동시 요청 (Java 21+) | Virtual Threads | 블로킹 코드 그대로, 확장성 확보 |
| 완전한 리액티브 시스템 | Spring WebFlux (Reactor) | 논-블로킹 I/O 전 스택 |
| CPU 집약적 연산 | 별도 스레드풀 (ForkJoin) | I/O 없는 계산은 비동기 이점 없음 |

> **정리**: JavaScript는 Event Loop으로 싱글 스레드가 비동기를 흉내내고, Java는 실제 멀티 스레드로 병렬 실행합니다.
> Java 21 이후로는 Virtual Threads 덕분에 블로킹 코드를 유지하면서도 Node.js 수준의 I/O 동시성을 달성할 수 있습니다.
