export function SessionLifecycleViz() {
  const [phase, setPhase] = useState('initial');
  const [log, setLog] = useState([]);
  const [reqCount, setReqCount] = useState(0);
  const logEndRef = useRef(null);

  const addLog = (msg, color) => {
    const time = new Date().toLocaleTimeString('ko-KR', { hour: '2-digit', minute: '2-digit', second: '2-digit' });
    setLog(prev => [...prev, { msg, color, time }]);
  };

  useEffect(() => {
    logEndRef.current?.scrollIntoView({ behavior: 'smooth' });
  }, [log]);

  const doLogin = () => {
    if (phase !== 'initial') return;
    setPhase('loggedIn');
    addLog('POST /login → 인증 성공 (user1 / ROLE_USER)', '#10b981');
    addLog('HttpSession 생성 → ID: abc123xyz', '#3b82f6');
    addLog('SPRING_SECURITY_CONTEXT 세션에 저장', '#8b5cf6');
    addLog('Set-Cookie: JSESSIONID=abc123xyz; Path=/; HttpOnly', '#f59e0b');
  };

  const doRequest = () => {
    if (phase !== 'loggedIn') return;
    const n = reqCount + 1;
    setReqCount(n);
    addLog(`GET /api/data (요청 #${n}) → Cookie: JSESSIONID=abc123xyz`, '#3b82f6');
    addLog('세션 조회 → SecurityContext 복원 → 인증된 요청 처리', '#10b981');
  };

  const doLogout = () => {
    if (phase !== 'loggedIn') return;
    setPhase('loggedOut');
    addLog('POST /logout → session.invalidate()', '#ef4444');
    addLog('SPRING_SECURITY_CONTEXT 세션에서 제거', '#ef4444');
    addLog('Set-Cookie: JSESSIONID=; Max-Age=0 (쿠키 삭제)', '#ef4444');
  };

  const doReset = () => {
    setPhase('initial');
    setLog([]);
    setReqCount(0);
  };

  const phaseInfo = {
    initial:   { label: '미인증', color: '#6b7280', cookie: '없음' },
    loggedIn:  { label: '인증됨', color: '#10b981', cookie: 'JSESSIONID=abc123xyz' },
    loggedOut: { label: '로그아웃', color: '#ef4444', cookie: '삭제됨' },
  };
  const info = phaseInfo[phase];

  return (
    <div style={{ fontFamily: 'system-ui, sans-serif', border: '1px solid var(--ifm-color-emphasis-300)', borderRadius: 12, overflow: 'hidden', marginBottom: 24 }}>
      <div style={{ padding: '14px 20px', borderBottom: '1px solid var(--ifm-color-emphasis-300)', display: 'flex', justifyContent: 'space-between', alignItems: 'center', flexWrap: 'wrap', gap: 8 }}>
        <strong style={{ fontSize: 15 }}>세션 생명주기 시뮬레이터</strong>
        <div style={{ display: 'flex', alignItems: 'center', gap: 8, fontSize: 13 }}>
          <div style={{ width: 8, height: 8, borderRadius: '50%', background: info.color }} />
          <span style={{ color: info.color, fontWeight: 600 }}>{info.label}</span>
          <span style={{ color: 'var(--ifm-color-emphasis-500)' }}>| Cookie: {info.cookie}</span>
        </div>
      </div>

      <div style={{ display: 'grid', gridTemplateColumns: '1fr 1fr', gap: 0 }}>
        <div style={{ padding: '16px', borderRight: '1px solid var(--ifm-color-emphasis-200)' }}>
          <div style={{ fontSize: 12, fontWeight: 600, color: 'var(--ifm-color-emphasis-600)', marginBottom: 10 }}>동작 시뮬레이션</div>
          <div style={{ display: 'flex', flexDirection: 'column', gap: 8 }}>
            {[
              { label: '① 로그인 (POST /login)', phase: 'initial', action: doLogin, color: '#10b981' },
              { label: '② 인증 요청 (GET /api/data)', phase: 'loggedIn', action: doRequest, color: '#3b82f6' },
              { label: '③ 로그아웃 (POST /logout)', phase: 'loggedIn', action: doLogout, color: '#ef4444' },
            ].map(({ label, phase: enabledPhase, action, color }) => (
              <button
                key={label}
                onClick={action}
                disabled={phase !== enabledPhase}
                style={{
                  padding: '10px', borderRadius: 8, border: 'none',
                  cursor: phase !== enabledPhase ? 'default' : 'pointer',
                  background: phase !== enabledPhase ? 'var(--ifm-color-emphasis-200)' : color,
                  color: phase !== enabledPhase ? 'var(--ifm-color-emphasis-400)' : '#fff',
                  fontSize: 13, fontWeight: 600, transition: 'all 0.2s',
                }}
              >{label}</button>
            ))}
            <button
              onClick={doReset}
              style={{ padding: '8px', borderRadius: 8, border: '1px solid var(--ifm-color-emphasis-300)', cursor: 'pointer', background: 'transparent', color: 'var(--ifm-color-emphasis-600)', fontSize: 13 }}
            >↺ 초기화</button>
          </div>

          <div style={{ marginTop: 14, padding: '12px', background: 'var(--ifm-color-emphasis-100)', borderRadius: 8, fontSize: 12 }}>
            <div style={{ fontWeight: 600, marginBottom: 6, color: 'var(--ifm-color-emphasis-700)' }}>HttpSession 상태</div>
            {phase === 'initial' && <div style={{ color: '#6b7280' }}>세션 없음 — 로그인 전</div>}
            {phase === 'loggedIn' && (
              <div style={{ color: 'var(--ifm-color-emphasis-700)', lineHeight: 1.7 }}>
                <span style={{ color: '#10b981', fontWeight: 600 }}>✓ 활성</span><br />
                ID: abc123xyz<br />
                KEY: SPRING_SECURITY_CONTEXT<br />
                USERNAME: user1 / ROLE: ROLE_USER<br />
                <span style={{ color: '#3b82f6' }}>처리된 요청: {reqCount}건</span>
              </div>
            )}
            {phase === 'loggedOut' && (
              <div>
                <div style={{ color: '#ef4444', fontWeight: 600 }}>✗ 무효화됨</div>
                <div style={{ color: 'var(--ifm-color-emphasis-500)', marginTop: 4, fontSize: 11, lineHeight: 1.6 }}>
                  invalidateHttpSession(true)<br />
                  deleteCookies("JSESSIONID")
                </div>
              </div>
            )}
          </div>
        </div>

        <div style={{ padding: '16px' }}>
          <div style={{ fontSize: 12, fontWeight: 600, color: 'var(--ifm-color-emphasis-600)', marginBottom: 10 }}>이벤트 로그</div>
          <div style={{ height: 270, overflowY: 'auto', display: 'flex', flexDirection: 'column', gap: 5 }}>
            {log.length === 0 && (
              <div style={{ color: 'var(--ifm-color-emphasis-400)', fontSize: 12, textAlign: 'center', marginTop: 60 }}>
                왼쪽 버튼으로 시뮬레이션을 시작하세요
              </div>
            )}
            {log.map((entry, i) => (
              <div key={i} style={{
                padding: '6px 10px', borderRadius: 6,
                background: 'var(--ifm-color-emphasis-100)',
                borderLeft: `3px solid ${entry.color}`,
                fontSize: 11, lineHeight: 1.4,
              }}>
                <span style={{ color: 'var(--ifm-color-emphasis-400)', marginRight: 6 }}>{entry.time}</span>
                <span>{entry.msg}</span>
              </div>
            ))}
            <div ref={logEndRef} />
          </div>
        </div>
      </div>
    </div>
  );
}

스프링 시큐리티는 **모든 요청이 컨트롤러에 닿기 전에 필터 체인을 통과한다**는 한 가지 전제에서 출발한다. 이 전제를 이해하면 CORS를 왜 컨트롤러가 아닌 시큐리티에서 처리해야 하는지, 로그인 실패를 왜 `@ControllerAdvice`로 잡지 못하는지, 세션이 언제 만들어지고 사라지는지 자연스럽게 이해된다.

## 1. 웹 요청 흐름: 필터 체인이 먼저다

HTTP 요청은 `DispatcherServlet`에 닿기 전에 `DelegatingFilterProxy` → `FilterChainProxy` → `SecurityFilterChain` 순서로 통과한다. 컨트롤러는 이 모든 관문을 통과한 요청만 받는다.

**DelegatingFilterProxy** 서블릿 컨테이너(Tomcat)는 스프링 빈을 직접 알 수 없다. DelegatingFilterProxy가 스프링 컨텍스트에 등록된 FilterChainProxy에 처리를 위임하는 브릿지 역할을 한다. 스프링 부트가 자동으로 등록하므로 직접 작성하지 않는다.

> 핵심: `@CrossOrigin`은 컨트롤러 어노테이션이다. 시큐리티 필터가 CORS Preflight를 먼저 막으면 컨트롤러까지 도달하지 못한다. CORS 설정은 반드시 `SecurityFilterChain`에서 처리해야 한다.

아래 시뮬레이터에서 각 필터를 클릭하거나 ▶ 버튼으로 전체 흐름을 확인하자.

<SecurityFilterChainViz />

| 필터 | 역할 | 적용 대상 |
|---|---|---|
| `CorsFilter` | Origin 검증, Preflight 응답 | 모든 요청 |
| `SecurityContextPersistenceFilter` | 세션 → SecurityContext 복원 | 모든 요청 |
| `UsernamePasswordAuthenticationFilter` | 로그인 처리 | POST `/login` 전용 |
| `ExceptionTranslationFilter` | 인증·인가 예외 → HTTP 변환 | 모든 요청 |
| `FilterSecurityInterceptor` | 인가 최종 결정 | 모든 요청 |

### 핵심 키워드
`DelegatingFilterProxy` `FilterChainProxy` `SecurityFilterChain` `CorsFilter` `FilterSecurityInterceptor`

## 2. DB 기반 인증 흐름

기본 인증(콘솔 임시 비밀번호)을 넘어 실무에서는 DB에 저장된 사용자 정보로 인증해야 한다. `UserDetailsService`를 구현하면 `UsernamePasswordAuthenticationFilter` → `AuthenticationManager` → `DaoAuthenticationProvider` → `UserDetailsService` → `BCryptPasswordEncoder` 로 이어지는 인증 체인이 완성된다.

개발자가 직접 비밀번호를 비교하지 않는 것이 핵심이다. `PasswordEncoder` 빈만 등록하면 `DaoAuthenticationProvider`가 내부에서 `matches()`를 호출해 검증한다.

<AuthFlowTabs />

```java
// SecurityFilterChain — UserDetailsService + PasswordEncoder 연결
@Bean
public SecurityFilterChain filterChain(HttpSecurity http,
        UserDetailsService userDetailsService) throws Exception {
    http
        .csrf(csrf -> csrf.disable())
        .cors(cors -> cors.configurationSource(corsConfigurationSource()))
        .formLogin(form -> form
            .loginProcessingUrl("/login")
            .successHandler(customSuccessHandler)
            .failureHandler(customFailureHandler)
        )
        .logout(logout -> logout
            .logoutUrl("/logout")
            .invalidateHttpSession(true)
            .deleteCookies("JSESSIONID")
        )
        .authorizeHttpRequests(auth -> auth
            .requestMatchers("/login", "/register").permitAll()
            .anyRequest().authenticated()
        )
        .userDetailsService(userDetailsService);
    return http.build();
}
```

### 핵심 키워드
`UserDetailsService` `UserDetails` `DaoAuthenticationProvider` `BCryptPasswordEncoder` `UsernameNotFoundException`

## 3. 세션 생명주기

로그인 성공 시 `HttpSession`이 생성되고 `JSESSIONID` 쿠키가 발급된다. 이후 브라우저는 모든 요청에 이 쿠키를 자동으로 포함하고, 서버는 세션에서 `SecurityContext`를 복원해 인증된 요청으로 처리한다.

**세션 타임아웃** 기본 30분이다. 마지막 요청으로부터 30분이 경과하면 세션이 만료된다. `server.servlet.session.timeout=1800` 으로 변경 가능하다.

<SessionLifecycleViz />

```properties
# application.properties
server.servlet.session.timeout=1800   # 초 단위 (기본 30분)

# 세션 쿠키 보안 옵션 (운영 환경)
server.servlet.session.cookie.http-only=true   # JS 접근 차단
server.servlet.session.cookie.secure=true      # HTTPS 전용
server.servlet.session.cookie.same-site=None   # 크로스 오리진 허용 시
```

> React + Spring 분리 아키텍처에서 세션 쿠키를 사용하려면 Axios에 `withCredentials: true`가 필요하다. 이 설정 없이는 브라우저가 크로스 오리진 요청에 쿠키를 포함하지 않는다.

### 핵심 키워드
`HttpSession` `JSESSIONID` `SecurityContextHolder` `invalidateHttpSession` `session.timeout`

## 4. React 연동: CORS · FormData · sessionStorage

분리 아키텍처(React + Spring Boot)에서 인증 연동 시 반드시 처리해야 할 세 가지 포인트가 있다.

### CORS — 시큐리티 필터에서 처리

```java
@Bean
CorsConfigurationSource corsConfigurationSource() {
    CorsConfiguration config = new CorsConfiguration();
    config.setAllowedOrigins(List.of("http://localhost:5173")); // Vite 개발 서버
    config.setAllowedMethods(List.of("GET", "POST", "PUT", "DELETE", "PATCH", "OPTIONS"));
    config.setAllowedHeaders(List.of("Authorization", "Content-Type", "Cache-Control"));
    config.setAllowCredentials(true); // 쿠키 포함 허용 — setAllowedOrigins("*")와 함께 쓸 수 없음

    UrlBasedCorsConfigurationSource source = new UrlBasedCorsConfigurationSource();
    source.registerCorsConfiguration("/**", config);
    return source;
}
```

### FormData — 시큐리티 기본 파라미터명 준수

스프링 시큐리티 폼 로그인은 기본적으로 `username`, `password` 파라미터를 기대한다. 프론트에서 다른 키 이름을 쓰면 인증 필터가 값을 읽지 못한다.

```javascript
// React — 로그인 요청
const loginHandler = async () => {
  const formData = new FormData();
  formData.append("username", user.username); // 시큐리티 기본값과 일치
  formData.append("password", user.password);
  if (rememberMe) formData.append("remember-me", "on");

  try {
    await axios.post("http://localhost:8080/login", formData, {
      withCredentials: true,
    });
    const me = await axios.get("http://localhost:8080/api/me", {
      withCredentials: true,
    });
    // sessionStorage에 사용자 정보 저장
    sessionStorage.setItem("user", JSON.stringify(me.data));
    setIsLoggedIn(true);
  } catch (error) {
    if (error.response?.status === 401) {
      setErrorMsg("아이디 또는 비밀번호를 확인하세요.");
    }
  }
};
```

### sessionStorage vs localStorage

**sessionStorage** 탭이 닫히면 자동 삭제된다. 로그인 상태 관리처럼 탭 종료 시 초기화되는 것이 자연스러운 데이터에 적합하다.
- `sessionStorage.setItem("user", JSON.stringify(data))`
- `JSON.parse(sessionStorage.getItem("user"))`

**localStorage** 브라우저를 닫아도 유지된다. 자동 로그인(rememberMe) 용도에 적합하지만, XSS에 취약하므로 민감한 토큰은 저장하지 않는다.

> 권한(`ROLE_USER`)은 시큐리티 특성상 `ROLE_` 접두사가 붙어 전달된다. 프론트에서 역할 기반 UI 분기 시 이 접두사를 고려해야 한다.

### 핵심 키워드
`withCredentials` `FormData` `sessionStorage` `ROLE_` 접두사 `allowCredentials`

## 정리: 세션 기반 vs 토큰(JWT) 기반

| 기준 | 세션(JSESSIONID) | JWT |
|---|---|---|
| 인증 정보 저장 위치 | 서버(세션 저장소) | 클라이언트(토큰 자체) |
| 서버 확장성 | 세션 동기화 필요 | Stateless — 수평 확장 유리 |
| 로그아웃 즉시성 | `invalidateHttpSession()` 즉시 무효화 | 만료까지 유효 (블랙리스트 별도 필요) |
| CSRF 위험 | 쿠키 기반이므로 CSRF 설정 필요 | Authorization 헤더 사용 시 CSRF 불필요 |
| 적합한 상황 | 단일 서버, 관리형 세션이 중요한 경우 | MSA, 수평 확장, 모바일 클라이언트 |

세션 기반이 나쁜 것이 아니다. 단일 서버 운영, 즉각적인 로그아웃 제어가 중요하거나 세션 관리 비용이 낮은 상황에서는 세션이 JWT보다 단순하고 안전하다. 아키텍처 맥락에 따라 선택하면 된다.
