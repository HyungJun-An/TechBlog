import { useState, useEffect, useRef } from 'react';

export function ScopeExplorer() {
  const [scenario, setScenario] = useState('model');
  const [step, setStep] = useState(-1);
  const [running, setRunning] = useState(false);
  const timerRef = useRef(null);

  const scenarios = {
    model: {
      label: 'Model (요청 스코프)',
      color: '#3b82f6',
      controller: `// 컨트롤러에서 model에 담기
model.addAttribute("loginVO", userVO);
return "board/list";`,
      template: `<!-- 타임리프: model 스코프 접근 -->
<span th:text="\${loginVO.name}">이름</span>`,
      steps: [
        { label: 'model.addAttribute("loginVO", userVO)', desc: '컨트롤러가 Model에 데이터를 담습니다.', ok: true },
        { label: 'return "board/list"', desc: '같은 요청 안에서 뷰 렌더링 — Model 데이터가 살아있습니다.', ok: true },
        { label: '${loginVO.name} 탐색', desc: 'Thymeleaf가 Model(요청 스코프)에서 loginVO를 찾습니다.', ok: true },
        { label: '렌더링 성공 ✅', desc: '데이터가 정상 출력됩니다.', ok: true },
      ],
      note: '리다이렉트(`redirect:`) 후에는 Model이 사라져 null이 됩니다. redirect 후에는 세션이나 RedirectAttributes를 쓰세요.',
    },
    session_ok: {
      label: '세션 → ${session.loginVO}',
      color: '#10b981',
      controller: `// 컨트롤러에서 세션에 담기
session.setAttribute("loginVO", userVO);
return "board/list";
// 또는 redirect: 후에도 유지됨`,
      template: `<!-- 타임리프: 세션 스코프 명시 접근 -->
<span th:text="\${session.loginVO.name}">이름</span>`,
      steps: [
        { label: 'session.setAttribute("loginVO", userVO)', desc: '세션에 저장 — 여러 요청에 걸쳐 유지됩니다.', ok: true },
        { label: 'redirect: 또는 새 요청', desc: '새 요청이 들어와도 세션은 살아있습니다.', ok: true },
        { label: '${session.loginVO.name} 탐색', desc: 'session 키워드로 세션 스코프를 명시해서 찾습니다.', ok: true },
        { label: '렌더링 성공 ✅', desc: '세션 데이터가 정상 출력됩니다.', ok: true },
      ],
      note: '`session.` 접두사가 핵심입니다. 없으면 Model에서만 탐색해 null이 됩니다.',
    },
    session_fail: {
      label: '세션 → ${loginVO} (❌ 실수)',
      color: '#ef4444',
      controller: `// 컨트롤러에서 세션에만 담기
session.setAttribute("loginVO", userVO);
// model에는 담지 않음
return "board/list";`,
      template: `<!-- 실수: session 접두사 없이 접근 -->
<span th:text="\${loginVO.name}">이름</span>`,
      steps: [
        { label: 'session.setAttribute("loginVO", userVO)', desc: '세션에만 저장 — Model에는 없습니다.', ok: true },
        { label: '${loginVO.name} 탐색 시작', desc: 'session 접두사가 없으므로 Model(요청 스코프)만 탐색합니다.', ok: false },
        { label: 'Model에서 loginVO 탐색', desc: 'Model에 담은 적이 없으므로 null 반환', ok: false },
        { label: '렌더링 실패 ❌', desc: 'null 출력 또는 NullPointerException 발생', ok: false },
      ],
      note: '가장 흔한 타임리프 실수입니다. 세션 데이터는 반드시 `${session.loginVO}` 형태로 접근하세요.',
    },
  };

  const sc = scenarios[scenario];

  const run = () => {
    if (running) return;
    setStep(-1);
    setRunning(true);
    let i = 0;
    const tick = () => {
      setStep(i++);
      if (i < sc.steps.length) timerRef.current = setTimeout(tick, 600);
      else setRunning(false);
    };
    timerRef.current = setTimeout(tick, 200);
  };

  useEffect(() => {
    clearTimeout(timerRef.current);
    setStep(-1);
    setRunning(false);
  }, [scenario]);

  useEffect(() => () => clearTimeout(timerRef.current), []);

  return (
    <div style={{ border: '1px solid var(--ifm-color-emphasis-300)', borderRadius: 10, overflow: 'hidden', marginBottom: 24 }}>
      <div style={{ padding: '12px 16px', background: 'var(--ifm-color-emphasis-100)', borderBottom: '1px solid var(--ifm-color-emphasis-300)' }}>
        <div style={{ fontWeight: 700, fontSize: 12, marginBottom: 8 }}>시나리오 선택:</div>
        <div style={{ display: 'flex', gap: 6, flexWrap: 'wrap' }}>
          {Object.entries(scenarios).map(([k, v]) => (
            <button key={k} onClick={() => setScenario(k)}
              style={{ padding: '5px 12px', borderRadius: 6, border: `1.5px solid ${v.color}`, background: scenario === k ? v.color : 'transparent', color: scenario === k ? '#fff' : v.color, cursor: 'pointer', fontWeight: 600, fontSize: 11 }}>
              {v.label}
            </button>
          ))}
        </div>
      </div>

      <div style={{ display: 'grid', gridTemplateColumns: '1fr 1fr', gap: 0, borderBottom: '1px solid var(--ifm-color-emphasis-200)' }}>
        <div style={{ padding: '12px 16px', borderRight: '1px solid var(--ifm-color-emphasis-200)' }}>
          <div style={{ fontSize: 11, fontWeight: 700, color: '#6b7280', marginBottom: 6 }}>Controller</div>
          <pre style={{ background: '#1e1e2e', color: '#cdd6f4', padding: 10, borderRadius: 8, fontSize: 11, margin: 0, overflowX: 'auto' }}><code>{sc.controller}</code></pre>
        </div>
        <div style={{ padding: '12px 16px' }}>
          <div style={{ fontSize: 11, fontWeight: 700, color: '#6b7280', marginBottom: 6 }}>Thymeleaf</div>
          <pre style={{ background: '#0f2417', color: '#86efac', padding: 10, borderRadius: 8, fontSize: 11, margin: 0, overflowX: 'auto' }}><code>{sc.template}</code></pre>
        </div>
      </div>

      <div style={{ padding: '10px 16px', display: 'flex', justifyContent: 'flex-end', gap: 8, borderBottom: '1px solid var(--ifm-color-emphasis-200)' }}>
        <button onClick={() => { clearTimeout(timerRef.current); setStep(-1); setRunning(false); }}
          style={{ padding: '4px 12px', borderRadius: 6, background: 'transparent', border: '1px solid var(--ifm-color-emphasis-400)', color: 'var(--ifm-font-color-base)', cursor: 'pointer', fontSize: 11 }}>
          초기화
        </button>
        <button onClick={run} disabled={running}
          style={{ padding: '6px 16px', borderRadius: 6, background: running ? '#6b7280' : 'var(--ifm-color-primary)', color: '#fff', border: 'none', cursor: running ? 'default' : 'pointer', fontWeight: 700, fontSize: 12 }}>
          {running ? '탐색 중...' : '▶ 스코프 탐색 실행'}
        </button>
      </div>

      <div style={{ padding: '12px 16px', display: 'flex', flexDirection: 'column', gap: 6 }}>
        {sc.steps.map((s, i) => {
          const active = step === i;
          const done = step > i;
          const c = s.ok ? sc.color : '#ef4444';
          return (
            <div key={i} style={{ display: 'flex', alignItems: 'flex-start', gap: 10, padding: '8px 12px', borderRadius: 8,
              background: active ? c + '15' : done ? 'var(--ifm-color-emphasis-100)' : 'transparent',
              border: `2px solid ${active ? c : done ? c + '55' : 'var(--ifm-color-emphasis-200)'}`,
              transition: 'all 0.25s' }}>
              <div style={{ width: 24, height: 24, borderRadius: '50%', flexShrink: 0, background: done || active ? c : 'var(--ifm-color-emphasis-200)', display: 'flex', alignItems: 'center', justifyContent: 'center', color: '#fff', fontWeight: 700, fontSize: 10 }}>
                {done ? (s.ok ? '✓' : '✗') : i + 1}
              </div>
              <div>
                <div style={{ fontWeight: 700, fontSize: 12, color: active ? c : 'var(--ifm-font-color-base)', fontFamily: 'monospace' }}>{s.label}</div>
                {(active || done) && <div style={{ fontSize: 11, color: 'var(--ifm-color-secondary-darkest)', marginTop: 2 }}>{s.desc}</div>}
              </div>
            </div>
          );
        })}
      </div>

      {step >= sc.steps.length - 1 && (
        <div style={{ margin: '0 16px 16px', padding: '10px 14px', borderRadius: 8,
          background: sc.color === '#ef4444' ? '#fee2e2' : '#d1fae5',
          border: `1px solid ${sc.color === '#ef4444' ? '#fca5a5' : '#6ee7b7'}` }}>
          <p style={{ margin: 0, fontSize: 12, color: sc.color === '#ef4444' ? '#991b1b' : '#065f46' }}>{sc.note}</p>
        </div>
      )}
    </div>
  );
}

    // ✅ 상태코드 + 바디 함께 반환

    public ResponseEntity<Map<String, Object>> checkId(
            @RequestParam String id) {

        Map<String, Object> body = new HashMap<>();
        body.put("isOk", userService.isIdAvailable(id));

        // ok() = 200 OK + 바디
        return ResponseEntity.ok(body);

    // ✅ 상황별 상태코드 분기
    @PostMapping("/user")
    public ResponseEntity<UserVO> createUser(@RequestBody UserVO user) {
        UserVO saved = userService.save(user);
        if (saved == null) {
            // 400 Bad Request, 바디 없음
            return ResponseEntity.badRequest().build();
        }
        // 201 Created + Location 헤더 + 바디
        URI location = URI.create("/api/user/" + saved.getId());
        return ResponseEntity.created(location).body(saved);
    }

    // ✅ 커스텀 헤더 추가

    public ResponseEntity<byte[]> download() {

## 타임리프 스코프 — <code>{'${loginVO}'}</code> vs <code>{'${session.loginVO}'}</code>

타임리프에서 가장 흔한 실수입니다. 세션에 저장된 값을 `session.` 없이 접근하면 null이 됩니다.

<ScopeExplorer />

### 스코프 비교 한눈에

| 표현식 | 탐색 위치 | 유지 범위 | 사용 예 |
|---|---|---|---|
| <code>{'${loginVO}'}</code> | Model(요청 스코프) | 요청 1회 | 컨트롤러가 model.addAttribute로 담은 데이터 |
| <code>{'${session.loginVO}'}</code> | HttpSession | 세션 만료까지 | 로그인 후 여러 페이지에 걸쳐 사용자 정보 유지 |
| <code>{'${param.keyword}'}</code> | 쿼리스트링 | 해당 요청만 | URL 파라미터 직접 접근 |

```java
// 컨트롤러 — 세션에 저장
@PostMapping("/login")
public String doLogin(HttpSession session,
                      @RequestParam Map<String, Object> params) {
    UserVO user = userService.login(params);
    if (user != null) {
        session.setAttribute("loginVO", user);   // 세션에 저장
        session.setMaxInactiveInterval(30 * 60); // 30분 타임아웃
        return "redirect:/board/list";
    }
    return "redirect:/user/login?error=true";
}
```

```html
<!-- 타임리프 — 세션 접근 -->
<th:block th:if="${session.loginVO != null}">
    <span th:text="${session.loginVO.name}">이름</span>님 환영합니다.
    <a href="/user/logout">로그아웃</a>
</th:block>

<th:block th:unless="${session.loginVO != null}">
    <a href="/user/login">로그인</a>
</th:block>
```

### `th:block` — 태그 없이 조건 처리

타임리프는 **속성 기반 템플릿**이라 조건을 붙일 태그가 반드시 필요합니다. 태그 없이 조건만 걸고 싶을 때 `th:block`을 씁니다. 렌더링 후 `<th:block>` 태그 자체는 사라집니다.

```html
<!-- ❌ 불필요한 <div> 추가 -->
<div th:if="${session.loginVO != null}">
    환영합니다!
</div>

<!-- ✅ th:block으로 DOM 오염 없이 -->
<th:block th:if="${session.loginVO != null}">
    환영합니다!
</th:block>
```

### `th:utext` — HTML 태그 포함 문자열 렌더링

```html
<!-- th:text: 이스케이프 처리 → <img> 문자 그대로 출력 -->
<td th:text="${vo.title}"></td>
<!-- 결과: &lt;img src="..."&gt; 제목텍스트 -->

<!-- th:utext: 이스케이프 없음 → 실제 <img> 태그로 렌더링 -->
<td th:utext="${vo.title}"></td>
<!-- 결과: 이미지가 실제로 표시됨 -->
```

> `th:utext`는 XSS 위험이 있습니다. **서버에서 생성한 신뢰할 수 있는 데이터**에만 사용하세요.

## ResponseEntity vs @ResponseBody

`@RestController`(= `@Controller + @ResponseBody`)가 JSON을 반환하는 기본 방법이라면, `ResponseEntity`는 HTTP 상태코드와 헤더까지 제어할 때 씁니다.

<ResponseEntityCompare />

### Gson으로 JSON 생성

`ResponseEntity` 없이 문자열로 JSON을 내려야 할 때 Gson을 씁니다.

```xml
<!-- pom.xml -->
<dependency>
    <groupId>com.google.code.gson</groupId>
    <artifactId>gson</artifactId>
</dependency>
```

```java
@RestController
public class UserApiController {

    // 방법 1: Gson으로 직접 문자열 생성
    @GetMapping("/check")
    public String checkIdGson(@RequestParam String id) {
        Map<String, Object> result = new HashMap<>();
        result.put("isOk", userService.isIdAvailable(id));
        return new GsonBuilder().create().toJson(result);
        // → {"isOk":true}
    }

    // 방법 2: ResponseEntity — Jackson이 자동 직렬화 (권장)
    @GetMapping("/check2")
    public ResponseEntity<Map<String, Object>> checkIdEntity(
            @RequestParam String id) {
        Map<String, Object> result = new HashMap<>();
        result.put("isOk", userService.isIdAvailable(id));
        return ResponseEntity.ok(result);
    }
}
```

> 스프링 부트는 Jackson이 기본 내장되어 있어 Map/VO를 반환하면 자동으로 JSON 직렬화합니다. Gson은 주로 레거시 환경이나 공공 API 파싱 시 사용됩니다.

`window.open`으로 팝업창을 열고, Ajax로 서버에 비동기 요청해 결과를 부모창에 반영합니다.

    opener.document.getElementById("userId").value = confirmedId;

    opener.document.getElementById("userId")

    opener.document.getElementById("userName").focus();

### Ajax vs 3가지 방식 비교

| 방식 | 의존성 | JSON 파싱 | HTTP 오류 처리 | 주요 사용처 |
|---|---|---|---|---|
| **jQuery Ajax** | jQuery 필요 | `dataType:"json"` 자동 | `error` 콜백 | 레거시, 빠른 프로토타입 |
| **fetch** | 없음 (브라우저 기본) | `response.json()` 수동 | `response.ok` 직접 확인 | 신규 프로젝트, 표준 |
| **axios** | npm/CDN 필요 | 자동 (`response.data`) | 4xx/5xx 자동 예외 | React/Vue SPA |

## 로그아웃: 세션 무효화 + 캐시 제거

```java
@GetMapping("/logout")
public String logout(HttpSession session, HttpServletResponse response) {
    // 세션 무효화 — 세션 데이터 전체 삭제
    session.invalidate();

    // 뒤로가기 시 캐시된 화면 노출 방지
    response.setHeader("Cache-Control", "no-cache, no-store, must-revalidate");
    response.setHeader("Pragma", "no-cache");
    response.setDateHeader("Expires", 0);

    return "redirect:/user/login";
}
```

> 캐시 제거는 인터셉터(`HandlerInterceptor`)에서 공통 처리하는 방식이 유지보수에 유리합니다. 인증이 필요한 페이지 전체에 일괄 적용할 수 있습니다.

## 한눈에 정리

| 기술 | 핵심 포인트 |
|---|---|
| **@SpringBootTest** | 스프링 컨텍스트 로딩 → @Autowired 주입 가능, 실제 DB 연동 검증 |
| **세션 스코프 접근** (`session.loginVO`) | 세션 스코프는 반드시 `session.` 접두사 명시, 없으면 Model에서만 탐색 |
| **th:block** | DOM 태그 없이 조건/반복 처리, 렌더링 후 태그 사라짐 |
| **th:utext** | HTML 태그 포함 문자열 렌더링, XSS 주의 |
| **@ResponseBody** | 반환값 → JSON, 상태코드는 200 고정 |
| **ResponseEntity** | 상태코드 + 헤더 + 바디 정밀 제어, REST API 설계 원칙에 부합 |
| **jQuery Ajax** | `dataType:"json"`으로 자동 파싱, 레거시 환경 적합 |
| **fetch** | 브라우저 기본 내장, `response.ok` 체크 필수 |
| **session.invalidate()** | 로그아웃 시 세션 무효화 + 캐시 헤더 함께 적용 |
