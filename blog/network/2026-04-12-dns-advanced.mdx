export function RoundRobinSimulator() {
  const [servers, setServers] = useState([
    { id: 1, ip: '203.0.113.10', label: 'Server A', alive: true, hitCount: 0 },
    { id: 2, ip: '203.0.113.20', label: 'Server B', alive: true, hitCount: 0 },
    { id: 3, ip: '203.0.113.30', label: 'Server C', alive: true, hitCount: 0 },
  ]);
  const [rrIndex, setRrIndex] = useState(0);
  const [log, setLog] = useState([]);
  const [animId, setAnimId] = useState(null);

  const query = () => {
    const idx = rrIndex % 3;
    const srv = servers[idx];
    const failed = !srv.alive;

    const entry = {
      id: Date.now(),
      ip: srv.ip,
      label: srv.label,
      failed,
      msg: failed
        ? `→ ${srv.ip} (${srv.label}) — 서버 다운! 사용자 접속 실패`
        : `→ ${srv.ip} (${srv.label}) — 응답 성공`,
    };

    setLog(prev => [entry, ...prev].slice(0, 6));
    setAnimId(srv.id);
    setRrIndex(prev => prev + 1);
    setServers(prev => prev.map(s =>
      s.id === srv.id ? { ...s, hitCount: s.hitCount + 1 } : s
    ));
    setTimeout(() => setAnimId(null), 500);
  };

  const toggleServer = (id) => {
    setServers(prev => prev.map(s => s.id === id ? { ...s, alive: !s.alive } : s));
  };

  const reset = () => {
    setServers(prev => prev.map(s => ({ ...s, alive: true, hitCount: 0 })));
    setRrIndex(0);
    setLog([]);
  };

  return (
    <div style={{ margin: '24px 0', fontFamily: 'monospace' }}>
      <div style={{ display: 'flex', gap: '16px', flexWrap: 'wrap' }}>
        {/* 서버 목록 */}
        <div style={{ flex: '1', minWidth: '200px' }}>
          <div style={{ fontSize: '12px', color: 'var(--ifm-color-emphasis-500)', marginBottom: '8px' }}>
            서버 클릭 = 다운/복구 토글
          </div>
          {servers.map((srv) => (
            <div
              key={srv.id}
              onClick={() => toggleServer(srv.id)}
              style={{
                display: 'flex', alignItems: 'center', gap: '10px',
                padding: '10px 14px', marginBottom: '6px', borderRadius: '8px',
                cursor: 'pointer', userSelect: 'none',
                border: `2px solid ${srv.alive ? '#27ae60' : '#e74c3c'}`,
                background: animId === srv.id
                  ? (srv.alive ? '#27ae6030' : '#e74c3c30')
                  : 'var(--ifm-card-background-color)',
                transition: 'background 0.3s',
                opacity: srv.alive ? 1 : 0.7,
              }}
            >
              <div style={{
                width: '10px', height: '10px', borderRadius: '50%',
                background: srv.alive ? '#27ae60' : '#e74c3c',
                flexShrink: 0,
              }} />
              <div style={{ flex: 1 }}>
                <div style={{ fontWeight: 700, fontSize: '13px' }}>{srv.label}</div>
                <div style={{ fontSize: '11px', color: 'var(--ifm-color-emphasis-500)' }}>{srv.ip}</div>
              </div>
              <div style={{ fontSize: '11px', color: 'var(--ifm-color-emphasis-600)', textAlign: 'right' }}>
                {srv.alive ? '정상' : '다운'}<br />
                <span style={{ fontSize: '10px' }}>요청 {srv.hitCount}회</span>
              </div>
            </div>
          ))}
          <div style={{ display: 'flex', gap: '8px', marginTop: '10px' }}>
            <button
              onClick={query}
              style={{
                flex: 1, padding: '10px', borderRadius: '8px', border: 'none',
                background: '#3498db', color: '#fff', fontWeight: 700,
                cursor: 'pointer', fontSize: '13px', fontFamily: 'monospace',
              }}
            >
              DNS 질의 →
            </button>
            <button
              onClick={reset}
              style={{
                padding: '10px 14px', borderRadius: '8px',
                border: '1px solid var(--ifm-color-emphasis-300)',
                background: 'transparent', cursor: 'pointer', fontSize: '12px',
                color: 'var(--ifm-color-emphasis-600)', fontFamily: 'monospace',
              }}
            >
              리셋
            </button>
          </div>
        </div>

        {/* 로그 */}
        <div style={{ flex: '1', minWidth: '220px' }}>
          <div style={{ fontSize: '12px', color: 'var(--ifm-color-emphasis-500)', marginBottom: '8px' }}>
            질의 로그 (최근 6건)
          </div>
          <div style={{
            background: '#0d1117', borderRadius: '8px', padding: '12px',
            minHeight: '160px', fontFamily: 'monospace',
          }}>
            {log.length === 0 ? (
              <div style={{ color: '#555', fontSize: '12px' }}>DNS 질의 버튼을 눌러보세요</div>
            ) : log.map((entry) => (
              <div key={entry.id} style={{
                fontSize: '12px', marginBottom: '5px',
                color: entry.failed ? '#e74c3c' : '#27ae60',
              }}>
                {entry.msg}
              </div>
            ))}
          </div>
          <div style={{
            marginTop: '10px', padding: '10px 12px', borderRadius: '8px',
            background: '#e67e2215', border: '1px solid #e67e2240',
            fontSize: '12px', lineHeight: 1.6, color: 'var(--ifm-color-emphasis-700)',
          }}>
            <b style={{ color: '#e67e22' }}>Round Robin 치명적 한계:</b><br />
            헬스 체크가 없어 서버가 다운돼도 계속 해당 IP를 응답합니다.<br />
            실무 해결책: AWS Route 53 / Cloudflare 같은 관리형 DNS 사용
          </div>
        </div>
      </div>
    </div>
  );
}

  const sel = selected ? protocols.find(p => p.id === selected) : null;

  return (
    <div style={{ margin: '24px 0', fontFamily: 'monospace' }}>
      <div style={{ display: 'flex', gap: '12px', flexWrap: 'wrap', marginBottom: '14px' }}>
        {protocols.map(p => (
          <div
            key={p.id}
            onClick={() => setSelected(selected === p.id ? null : p.id)}
            style={{
              flex: '1', minWidth: '140px', padding: '14px',
              borderRadius: '10px', cursor: 'pointer', userSelect: 'none',
              border: `2px solid ${p.color}`,
              background: selected === p.id ? p.color + '18' : 'var(--ifm-card-background-color)',
              outline: selected === p.id ? `3px solid ${p.color}` : 'none',

            }}
          >
            <div style={{ fontSize: '22px', marginBottom: '4px' }}>{p.icon}</div>

            {p.en && <div style={{ fontSize: '10px', color: 'var(--ifm-color-emphasis-500)', marginTop: '2px' }}>{p.en}</div>}

              <div>식별: <b>{p.identifiable}</b></div>
              <div>적용: <b>{p.level}</b></div>
            </div>
          </div>
        ))}
      </div>

      {sel && (
        <div style={{

          background: sel.color + '10', border: `2px solid ${sel.color}`,

        </div>
      )}

      <div style={{
        marginTop: '14px', padding: '12px 14px', borderRadius: '8px',
        background: '#27ae6015', border: '1px solid #27ae6040',

      }}>
        <b style={{ color: '#27ae60' }}>DNSSEC vs DoH/DoT:</b><br />

        DoH/DoT = 질의 <b>기밀성 보장</b> (도청 방지)<br />

      </div>
    </div>
  );
}

## 1. DNSSEC — 신뢰 체인으로 응답을 검증하다

DNS 응답은 기본적으로 검증 방법이 없습니다. 공격자가 `example.com → 192.168.1.1` 이라는 가짜 응답을 캐시에 심으면 (**캐시 포이즈닝**), 클라이언트는 이를 구분할 방법이 없습니다.

**DNSSEC**은 공개키 암호화 기반 디지털 서명으로 이 문제를 해결합니다:

- **RRSIG** — DNS 레코드셋에 대한 디지털 서명
- **DNSKEY** — 서명 검증에 사용하는 공개키 (KSK: 키 서명 키, ZSK: 존 서명 키)
- **DS** — 자식 존의 DNSKEY를 해시한 위임 서명자 레코드 (부모 존에 등록)

루트 존부터 최종 도메인까지 체인이 이어져야 전체 검증이 성립합니다. 리졸버에 사전 내장된 루트 존 공개키(**Trust Anchor**)가 신뢰의 시작점입니다.

<DNSSECChainViz />

:::info KSK와 ZSK를 분리하는 이유
KSK(2048비트)는 거의 바뀌지 않는 최상위 키, ZSK(1024비트)는 주기적으로 롤오버합니다. 비용이 큰 키 교체를 분리해 관리 부담을 줄이기 위한 설계입니다.
:::

## 2. DNS Round Robin — 헬스 체크 없는 부하 분산

하나의 도메인에 여러 A 레코드를 등록하면, DNS 서버는 질의마다 순서를 돌아가며 다른 IP를 응답합니다. 추가 인프라 없이 **비용 0원**으로 부하를 분산할 수 있습니다.

```
example.com  A  203.0.113.10   ; Server A
example.com  A  203.0.113.20   ; Server B
example.com  A  203.0.113.30   ; Server C
```

단, 치명적인 한계가 있습니다. **서버를 클릭해 다운시키고 DNS 질의를 계속 보내보세요:**

<RoundRobinSimulator />

서버가 다운돼도 DNS는 해당 IP를 계속 응답합니다. 브라우저의 **Happy Eyeballs** 알고리즘이 부분적으로 완화해주지만, 근본 해결책은 아닙니다.

| 항목 | Round Robin | 관리형 DNS (Route 53 등) |
|------|-------------|--------------------------|
| 비용 | 0원 | 유료 |
| 헬스 체크 | 없음 | 있음 |
| 세션 유지 | 불가 | 설정 가능 |
| 가중치 분산 | 불가 | 가능 |

실무에서는 단독 사용보다 **AWS Route 53, Cloudflare DNS** 같은 관리형 서비스로 헬스 체크와 가중치 라우팅을 결합해 사용합니다.

## 3. CDN과 DNS — 가장 가까운 엣지로 안내하기

CDN은 전 세계 엣지 서버에 콘텐츠를 캐시해두고, DNS를 통해 클라이언트를 **가장 가까운 서버**로 유도합니다. 적용 자체는 간단합니다:

```
www.example.com  CNAME  www.example.com.cdn-provider.net
```

CNAME만 바꾸면 이후 지역 기반 라우팅은 CDN 제공자가 처리합니다. 라우팅 방식은 크게 두 가지입니다:

<CDNRoutingViz />

- **GeoDNS**: 클라이언트 IP → 지리 DB 조회 → 가까운 엣지 IP 응답. IP 기반이라 VPN으로 우회 가능
- **Anycast**: 동일 IP를 여러 엣지가 공유, BGP가 네트워크 토폴로지상 최적 노드로 자동 유도. Cloudflare, AWS CloudFront가 채택

:::tip CDN 전환 후 캐시 지연
TTL이 남아있는 동안 구 IP를 가리킬 수 있습니다. 급하게 전환해야 할 때는 전환 전 TTL을 낮춰두세요 (e.g. 300초 → 60초).
:::

## 4. DoH / DoT — 질의 자체를 암호화하다

기존 DNS는 UDP 53번 포트로 **평문** 전송됩니다. ISP, 공공 Wi-Fi, 중간 라우터가 어떤 도메인을 조회하는지 모두 볼 수 있습니다. DNSSEC이 응답 **위변조**를 막는다면, DoH/DoT는 질의 **도청**을 막습니다.

<DNSPrivacyComparison />

실무 적용 기준:
- **서버-리졸버 구간**: DoT (전용 포트 853, 인프라 수준 설정)
- **엔드유저 프라이버시**: DoH (443, 브라우저 직접 지원 — Firefox/Chrome)
- **검열 우회 환경**: DoH (HTTPS 트래픽과 구분 불가)

## 정리 — 네 가지를 한 문장으로

| 기술 | 해결하는 문제 | 핵심 메커니즘 |
|------|-------------|--------------|
| DNSSEC | 응답 위변조 | 공개키 서명 + 신뢰 체인 |
| Round Robin | 단순 부하 분산 | 다중 A 레코드 순환 응답 |
| GeoDNS / Anycast | 지연 최소화 | IP 기반 지역 라우팅 / BGP 경로 선택 |
| DoH / DoT | 질의 도청 방지 | TLS/HTTPS 암호화 |

DNSSEC과 DoH/DoT는 서로 **보완 관계**입니다. DNSSEC은 "응답이 진짜인가", DoH/DoT는 "질의가 누군가에게 보이지 않는가"를 각각 해결합니다. 보안을 진지하게 고려한다면 두 가지를 함께 적용하는 것이 정석입니다.
